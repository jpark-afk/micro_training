CMAKE_MINIMUM_REQUIRED(VERSION 3.12)

#===============================================================================
# Generate file with DDS Micro Build ID
#===============================================================================

IF (${HAS_STRING_TIMESTAMP})
    string(TIMESTAMP generation_timestamp "%Y%m%dT%H%M%SZ" UTC)
ELSE()
    set(generation_timestamp "unsupported in cmake < 2.8.11")
ENDIF()

set(RTIME_BUILD_ID ${generation_timestamp})

FILE(WRITE ${BUILD_ID_FILE} "#define RTIME_BUILD_ID \"${RTIME_BUILD_ID}\"")

IF (${CMAKE_HOST_SYSTEM} MATCHES "Windows")
    FILE(APPEND ${BUILD_ID_FILE} "\r\n")
ELSE()
    FILE(APPEND ${BUILD_ID_FILE} "\n")
ENDIF()
