SET(RTIME_CERT_DOC
"Checking this option builds Micro as a single static library with only the Micro Cert APIs enabled")

IF (RTIME_CERT)
    SET(RTIME_CERT TRUE CACHE BOOL ${RTIME_CERT_DOC})
    GET_PROPERTY(rti_cert_flags DIRECTORY PROPERTY 
                 COMPILE_DEFINITIONS)
    LIST(LENGTH rti_cert_flags rti_cert_length)

    # Unclear if rti_cert_flags can ever be < 0 
    IF (${rti_cert_length} GREATER -1)
        LIST(FIND rti_cert_flags RTI_CERT rti_cert_index)
        IF (${rti_cert_index} EQUAL -1)
            SET_PROPERTY(DIRECTORY APPEND PROPERTY 
                         COMPILE_DEFINITIONS
                         RTI_CERT)
            GET_PROPERTY(rti_cert_flags DIRECTORY PROPERTY
                        COMPILE_DEFINITIONS)
        ENDIF()
    ENDIF()
ELSE()
    SET(RTIME_CERT FALSE CACHE BOOL ${RTIME_CERT_DOC})
ENDIF()

MACRO(RTIME_SET_CERT_DEFAULT name cert_default default docstring)
    IF (RTIME_CERT AND (NOT DEFINED ${name}))
        SET(${name} ${cert_default} CACHE BOOL ${docstring})
    ELSEIF (NOT DEFINED ${name})
        SET(${name} ${default} CACHE BOOL ${docstring})
    ELSE()
        SET(${name} ${${name}} CACHE BOOL ${docstring}})
    ENDIF()
ENDMACRO()
