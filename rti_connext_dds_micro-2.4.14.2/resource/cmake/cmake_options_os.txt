IF (RTIME_PRINT_CONFIG)
    RTI_MESSAGE_STATUS("RTIME_OSAPI_PLATFORM ...............: ${RTIME_OSAPI_PLATFORM}")
    RETURN()
ENDIF()

#== Look for available platforms
SET(RTIME_OSAPI_MODULE "osapi.2.0")

GET_MODULE_DIR(${RTIME_OSAPI_MODULE} osdir)

SET(RTIME_OSAPI_AUTO    "<auto detect>")
SET(RTIME_OSAPI_DEFAULT ${RTIME_OSAPI_AUTO})
SET(RTIME_OSAPI_POSIX     "posix")
SET(RTIME_OSAPI_VXWORKS   "vxworks")
SET(RTIME_OSAPI_AUTOSAR   "autosar")
SET(RTIME_OSAPI_WINDOWS   "windows")
SET(RTIME_OSAPI_FREERTOS  "freertos")
SET(RTIME_OSAPI_THREADX   "threadx")

IF (DEFINED _RTIME_OSAPI_PLATFORM)
    SET(RTIME_OSAPI_DEFAULT "${_RTIME_OSAPI_PLATFORM}")
ENDIF()

SET(RTIME_OSAPI_PLATFORM ${RTIME_OSAPI_DEFAULT} CACHE STRING "Target platform")

LIST(APPEND RTIME_OSAPI_PLATFORM_LIST ${RTIME_OSAPI_AUTO})

IF (${RTIME_DEVTREE})
    FILE(GLOB platforms LIST_DIRECTORIES FALSE
         ${RTIME_SOURCE_ROOT}/${osdir}/include/osapi/*osapi_os_*.h)
ELSE()
    FILE(GLOB platforms LIST_DIRECTORIES FALSE
         ${RTIME_SOURCE_ROOT}/include/osapi/*osapi_os_*.h)
ENDIF()

FOREACH(x ${platforms})
        GET_FILENAME_COMPONENT(osname ${x} NAME)
        STRING(REGEX REPLACE ".*osapi_os_(.*).h" "\\1" os ${osname})
        LIST(APPEND RTIME_OSAPI_PLATFORM_LIST ${os})
        SET(${os}_FILE ${osname})
ENDFOREACH()

SET_PROPERTY(CACHE RTIME_OSAPI_PLATFORM PROPERTY
             STRINGS ${RTIME_OSAPI_PLATFORM_LIST})

IF (NOT ${RTIME_OSAPI_PLATFORM} MATCHES ${RTIME_OSAPI_AUTO})
    IF(${RTIME_OSAPI_PLATFORM} MATCHES "(.*):(.*)")
        SET(RTIME_OSAPI_PLATFORM ${CMAKE_MATCH_1})
        SET(RTIME_OSAPI_PLATFORM_INCLUDE "osapi_os_${CMAKE_MATCH_2}.h")
    ELSE()
        SET(RTIME_OSAPI_PLATFORM ${CMAKE_MATCH_1})
        SET(RTIME_OSAPI_PLATFORM_INCLUDE ${${RTIME_OSAPI_PLATFORM}_FILE})
    ENDIF()
    ADD_DEFINITIONS(-DOSAPI_OS_DEF_H=${RTIME_OSAPI_PLATFORM_INCLUDE})
ENDIF()
