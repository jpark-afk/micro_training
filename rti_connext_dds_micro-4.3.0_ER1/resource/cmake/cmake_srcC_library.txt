SET(RTI_EXPORT_SUBMODULE_OUTPUT_BIN ${RTI_MODULE_OUTPUT_SRC}/lib/${RTIME_TARGET_NAME})
SET(RTI_EXPORT_SUBMODULE_OUTPUT_LIB ${RTI_MODULE_OUTPUT_SRC}/lib/${RTIME_TARGET_NAME})
SET(RTI_EXPORT_SUBMODULE_OUTPUT_PDF ${RTI_MODULE_OUTPUT_SRC}/lib/${RTIME_TARGET_NAME})
SET(RTI_EXPORT_SUBMODULE_OUTPUT_SRC ${RTI_MODULE_OUTPUT_SRC})
SET(RTI_EXPORT_SUBMODULE_OUTPUT_INCLUDE ${RTI_MODULE_OUTPUT_SRC}/include)

FOREACH(srcdir "" /srcC /srcCxx)
    IF (EXISTS ${RTI_MODULE_SRCROOT}${srcdir}/${RTI_SUBMODULE_NAME})
        SET(SRC_ROOT "${RTI_MODULE_SRCROOT}${srcdir}/${RTI_SUBMODULE_NAME}")

        FOREACH(f ${RTI_SUBMODULE_SRC})
            LIST(APPEND ${RTI_SUBMODULE_NAME}_SRC_C ${SRC_ROOT}/${f})
        ENDFOREACH()

        FOREACH(f ${RTI_SUBMODULE_HDR})
            LIST(APPEND ${RTI_SUBMODULE_NAME}_SRC_H ${SRC_ROOT}/${f})
        ENDFOREACH()

        IF (NOT ${srcdir} STREQUAL "")
            STRING(REPLACE "/" "" srcdirg ${srcdir})
        ENDIF()

        SOURCE_GROUP(\\${RTI_MODULE_NAME}\\${srcdirg}\\${RTI_SUBMODULE_NAME}
                 FILES ${${RTI_MODULE_NAME}_${RTI_SUBMODULE_NAME}_SRC}
                       ${${RTI_MODULE_NAME}_${RTI_SUBMODULE_NAME}_HDR}
                       ${RTI_SUBMODULE_IDL})

        LIST(LENGTH RTI_SUBMODULE_IDL idl_count)
        IF (${idl_count} GREATER 0)
            SET(IDL_OUTPUT_DIR ${RTI_MODULE_OUTPUT_SRC}${srcdir}/${RTI_SUBMODULE_NAME})
            LIST(APPEND RTI_MODULE_INCLUDE_DIRS ${IDL_OUTPUT_DIR})
            MICRO_ADD_IDL(IDL_GEN_C IDL_GEN_H RTI_SUBMODULE_IDL SRC_ROOT ${srcdir} ${IDL_OUTPUT_DIR})
            LIST(APPEND ${RTI_SUBMODULE_NAME}_SRC_C  ${IDL_GEN_C})
            LIST(APPEND ${RTI_SUBMODULE_NAME}_SRC_H  ${IDL_GEN_H})
            LIST(APPEND ${RTI_MODULE_NAME}_${RTI_SUBMODULE_NAME}_SRC_IDL  ${IDL_GEN_H} ${IDL_GEN_C})
            SET_PROPERTY(SOURCE  ${IDL_GEN_H} ${IDL_GEN_C}
                         PROPERTY RTI_GENERATED true)

        ENDIF()

        LIST(LENGTH RTI_SUBMODULE_XML_APPGEN xml_count)
        IF (${xml_count} GREATER 0)
            SET(XML_APPGEN_OUTPUT_DIR ${RTI_MODULE_OUTPUT_SRC}${srcdir}/${RTI_SUBMODULE_NAME})
            LIST(APPEND RTI_MODULE_INCLUDE_DIRS ${XML_APPGEN_OUTPUT_DIR})
            MICRO_ADD_XML_APPGEN(XML_APPGEN_GEN_C XML_APPGEN_GEN_H RTI_SUBMODULE_XML_APPGEN SRC_ROOT ${srcdir} ${XML_APPGEN_OUTPUT_DIR})
            LIST(APPEND ${RTI_SUBMODULE_NAME}_SRC_C  ${XML_APPGEN_GEN_C})
            LIST(APPEND ${RTI_SUBMODULE_NAME}_SRC_H  ${XML_APPGEN_GEN_H})
            LIST(APPEND ${RTI_MODULE_NAME}_${RTI_SUBMODULE_NAME}_SRC_XML_APPGEN  ${XML_APPGEN_GEN_H} ${XML_APPGEN_GEN_C})
            SET_PROPERTY(SOURCE  ${XML_APPGEN_GEN_H} ${XML_APPGEN_GEN_C}
                         PROPERTY RTI_GENERATED true)

        ENDIF()
    ENDIF()
ENDFOREACH()

FOREACH(idl ${RTI_MODULE_PUBLIC_HDR_IDL_EXPORT})
        LIST(APPEND ${RTI_MODULE_NAME}_${RTI_SUBMODULE_NAME}_IDL_EXPORT
                    ${IDL_OUTPUT_DIR}/${idl}.h
                    ${IDL_OUTPUT_DIR}/${idl}Plugin.h
                    ${IDL_OUTPUT_DIR}/${idl}Support.h)
ENDFOREACH()

SET(RTI_SUBMODULE_LIBNAME_SHARED ${RTI_LIB_PREFIX}_${RTI_SUBMODULE_NAME}${RTI_LIB_SUFFIX_DYNAMIC})
SET(RTI_SUBMODULE_LIBNAME_STATIC ${RTI_LIB_PREFIX}_${RTI_SUBMODULE_NAME}${RTI_LIB_SUFFIX_STATIC})

#MESSAGE("RTI_MODULE_INCLUDE_DIRS = ${RTI_MODULE_INCLUDE_DIRS}")
#MESSAGE("Adding ${RTI_LIB_PREFIX}_${RTI_SUBMODULE_NAME}${RTI_LIB_SUFFIX_DYNAMIC} and ${RTI_LIB_PREFIX}_${RTI_SUBMODULE_NAME}${RTI_LIB_SUFFIX_STATIC}")

ADD_LIBRARY(${RTI_SUBMODULE_LIBNAME_SHARED} SHARED
            ${${RTI_SUBMODULE_NAME}_SRC_C} ${${RTI_SUBMODULE_NAME}_SRC_H}
            ${${RTI_MODULE_NAME}_${RTI_SUBMODULE_NAME}_SRC_IDL})

FOREACH(mdin ${RTI_MODULE_DEPENDS} ${RTI_MODULE_IMPORTS})
    IF (${mdin} MATCHES "(.*)::(.*)")
        SET(mdin "${CMAKE_MATCH_1}")
    ENDIF()
    SET_PROPERTY(TARGET ${RTI_SUBMODULE_LIBNAME_SHARED} APPEND
                 PROPERTY COMPILE_DEFINITIONS RTI_${mdin}_DLL_VARIABLE)

    SET_TARGET_PROPERTIES(${RTI_SUBMODULE_LIBNAME_SHARED}
                        PROPERTIES CMAKE_SKIP_RPATH TRUE)
ENDFOREACH()

ADD_LIBRARY(${RTI_SUBMODULE_LIBNAME_STATIC} STATIC
            ${${RTI_SUBMODULE_NAME}_SRC_C} ${${RTI_SUBMODULE_NAME}_SRC_H}
            ${${RTI_MODULE_NAME}_${RTI_SUBMODULE_NAME}_SRC_IDL})


LIST(LENGTH RTI_SUBMODULE_IDL x)
IF (${x} GREATER 0)
    ADD_DEPENDENCIES(${RTI_SUBMODULE_LIBNAME_SHARED}
                     ${RTI_SUBMODULE_LIBNAME_STATIC})
ENDIF()

IF (${HAS_TARGET_INCLUDE_DIR})
   TARGET_INCLUDE_DIRECTORIES(${RTI_SUBMODULE_LIBNAME_SHARED}
                              PRIVATE ${RTI_MODULE_INCLUDE_DIRS})
   TARGET_INCLUDE_DIRECTORIES(${RTI_SUBMODULE_LIBNAME_STATIC}
                              PRIVATE ${RTI_MODULE_INCLUDE_DIRS})
ELSE()
    INCLUDE_DIRECTORIES(${RTI_MODULE_INCLUDE_DIRS})
ENDIF()

FOREACH(md ${RTI_MODULE_IMPORTS} ${RTI_LIBRARY_DEPENDS} ${RTI_SUBMODULE_LIBRARY_DEPENDS})
    add_exported_header_dependency(DEPEDENCY
                                        ${md}
                                   DEPENDENTS
                                        ${RTI_SUBMODULE_LIBNAME_SHARED}
                                        ${RTI_SUBMODULE_LIBNAME_STATIC}
    )
    add_exported_header_dependency(DEPEDENCY
                                        ${md}
                                   DEPENDENTS
                                        ${RTI_SUBMODULE_LIBNAME_SHARED}
                                        ${RTI_SUBMODULE_LIBNAME_STATIC}
    )
    IF (DEFINED ${md}_EXPORT_NAME)
        LIST(APPEND LIB_DEPS_STATIC  ${${md}_EXPORT_NAME}${RTI_LIB_SUFFIX_STATIC} )
        LIST(APPEND LIB_DEPS_DYNAMIC ${${md}_EXPORT_NAME}${RTI_LIB_SUFFIX_DYNAMIC} )
    ELSE()
        LIST(APPEND LIB_DEPS_STATIC ${RTI_LIB_PREFIX}_${md}${RTI_LIB_SUFFIX_STATIC} )
        LIST(APPEND LIB_DEPS_DYNAMIC ${RTI_LIB_PREFIX}_${md}${RTI_LIB_SUFFIX_DYNAMIC} )
    ENDIF()
ENDFOREACH()

SET_PROPERTY(TARGET ${RTI_SUBMODULE_LIBNAME_STATIC}  APPEND
             PROPERTY COMPILE_DEFINITIONS
             RTIME_TARGET_NAME=\"${RTIME_TARGET_NAME}\"
)

TARGET_LINK_LIBRARIES(${RTI_SUBMODULE_LIBNAME_STATIC} PRIVATE
                      ${LIB_DEPS_STATIC} ${PLATFORM_LIBS})

# This is just to test if this is a windows build, not if
# this is a shared library.
IF (${CMAKE_SHARED_LIBRARY_SUFFIX} STREQUAL ".dll")
    SET(libsuffix "lib")
    IF (${RTI_DEBUG_BUILD})
        LIST(APPEND libsuffix "pdb")
    ENDIF()
    FOREACH(f ${libsuffix})
        ADD_CUSTOM_COMMAND(TARGET ${RTI_SUBMODULE_LIBNAME_STATIC} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy
            $<TARGET_LINKER_FILE_DIR:${RTI_SUBMODULE_LIBNAME_STATIC}>/${CMAKE_STATIC_LIBRARY_PREFIX}${RTI_SUBMODULE_LIBNAME_STATIC}.${f}
            ${RTI_EXPORT_SUBMODULE_OUTPUT_LIB}/${CMAKE_STATIC_LIBRARY_PREFIX}${RTI_SUBMODULE_LIBNAME_STATIC}.${f}
        )
    ENDFOREACH()
ELSE()
    ADD_CUSTOM_COMMAND(TARGET ${RTI_SUBMODULE_LIBNAME_STATIC} POST_BUILD
              COMMAND ${CMAKE_COMMAND} -E make_directory
                      ${RTI_EXPORT_SUBMODULE_OUTPUT_LIB}
               COMMAND ${CMAKE_COMMAND} -E copy
               $<TARGET_FILE:${RTI_SUBMODULE_LIBNAME_STATIC}>
                           ${RTI_EXPORT_SUBMODULE_OUTPUT_LIB}/${CMAKE_STATIC_LIBRARY_PREFIX}${RTI_SUBMODULE_LIBNAME_STATIC})
ENDIF()

# To support not building shared libraries at test needs to be
# done on this variable. NOT (DEFINED RTI_NO_SHARED_LIB)
IF (TRUE)
    TARGET_LINK_LIBRARIES(${RTI_SUBMODULE_LIBNAME_SHARED} PRIVATE
                      ${LIB_DEPS_DYNAMIC} ${PLATFORM_LIBS})

    # For windows we need to copy .exp,*.lib., *.pdb and *.dll
    IF (${CMAKE_SHARED_LIBRARY_SUFFIX} STREQUAL ".dll")
        SET(libsuffix "dll" "lib" "exp")
        IF (${RTI_DEBUG_BUILD})
            LIST(APPEND libsuffix "pdb")
        ENDIF()
        FOREACH(f ${libsuffix})
           ADD_CUSTOM_COMMAND(TARGET ${RTI_SUBMODULE_LIBNAME_SHARED} POST_BUILD
                       COMMAND ${CMAKE_COMMAND} -E copy
                       $<TARGET_LINKER_FILE_DIR:${RTI_SUBMODULE_LIBNAME_SHARED}>/${CMAKE_SHARED_LIBRARY_PREFIX}${RTI_SUBMODULE_LIBNAME_SHARED}.${f}
                       ${RTI_EXPORT_SUBMODULE_OUTPUT_LIB}/${CMAKE_SHARED_LIBRARY_PREFIX}${RTI_SUBMODULE_LIBNAME_SHARED}.${f})
        ENDFOREACH()
    ELSE()
        ADD_CUSTOM_COMMAND(TARGET ${RTI_SUBMODULE_LIBNAME_SHARED} POST_BUILD
                   COMMAND ${CMAKE_COMMAND} -E copy
                   $<TARGET_LINKER_FILE:${RTI_SUBMODULE_LIBNAME_SHARED}>
                   ${RTI_EXPORT_SUBMODULE_OUTPUT_LIB}/${CMAKE_SHARED_LIBRARY_PREFIX}${RTI_SUBMODULE_LIBNAME_SHARED})
    ENDIF()
ENDIF()

FOREACH(mdin ${RTI_MODULE_IMPORTS} ${RTI_MODULE_DEPENDS}
${RTI_SUBMODULE_LIBRARY_DEPENDS} ${RTI_DLL_LIBRARY})
    STRING(REGEX REPLACE "lib(.*)" "\\1" md ${mdin})
    IF (${md} MATCHES "(.*)::(.*)")
        SET(md "${CMAKE_MATCH_1}")
    ENDIF()
    SET_PROPERTY(TARGET ${RTI_SUBMODULE_LIBNAME_SHARED} APPEND
    PROPERTY COMPILE_DEFINITIONS RTI_${md}_DLL_VARIABLE)
    IF (DEFINED ${md}_EXPORT_NAME)
    SET_PROPERTY(TARGET ${RTI_SUBMODULE_LIBNAME_SHARED} APPEND
    PROPERTY COMPILE_DEFINITIONS RTI_${${md}_EXPORT_NAME}_DLL_VARIABLE)
    ENDIF()

    IF (DEFINED ${md}_OBJECT_DEPENDS)
        FOREACH(mdin2 ${${md}_OBJECT_DEPENDS})
        IF (${mdin2} MATCHES "(.*)::(.*)")
            SET(mdin2 "${CMAKE_MATCH_1}")
        ENDIF()
        SET_PROPERTY(TARGET ${RTI_SUBMODULE_LIBNAME_SHARED} APPEND
                PROPERTY COMPILE_DEFINITIONS RTI_${mdin2}_DLL_VARIABLE)
        ENDFOREACH()
    ENDIF()
ENDFOREACH()

SET_TARGET_PROPERTIES(${RTI_SUBMODULE_LIBNAME_SHARED}
              PROPERTIES
              MACOSX_RPATH TRUE
)

             PROPERTY COMPILE_DEFINITIONS
             RTIME_TARGET_NAME=\"${RTIME_TARGET_NAME}\"
             RTI_${RTI_MODULE_NAME}_DLL_EXPORT
             RTI_${RTI_MODULE_NAME}_DLL_VARIABLE
)

FOREACH(srcdir "" /srcC /srcCxx)
    IF (EXISTS ${RTI_MODULE_SRCROOT}${srcdir}/${RTI_SUBMODULE_NAME}/test/CMakeLists.txt)
        SET(SRC_ROOT "${RTI_MODULE_SRCROOT}${srcdir}/${RTI_SUBMODULE_NAME}/test")
        INCLUDE(${SRC_ROOT}/CMakeLists.txt)

        FOREACH(f ${RTI_SUBMODULE_TEST_SRC} ${RTI_SUBMODULE_TEST_SRC_CC})
            LIST(APPEND ${RTI_MODULE_NAME}_TEST_SRC_C ${SRC_ROOT}/${f})
        ENDFOREACH()

        FOREACH(f ${RTI_SUBMODULE_TEST_HDR})
            LIST(APPEND ${RTI_MODULE_NAME}_TEST_SRC_H ${SRC_ROOT}/${f})
        ENDFOREACH()

    ENDIF()
ENDFOREACH()

SET(SM_HAS_GENERATED_FILES FALSE)
SET(${RTI_SUBMODULE_NAME}_GENERATED)

FOREACH(f ${${RTI_SUBMODULE_NAME}_SRC_C} ${${RTI_SUBMODULE_NAME}_SRC_H})
    GET_PROPERTY(rti_generated
                 SOURCE ${f}
                 PROPERTY RTI_GENERATED
                 SET)
    IF (rti_generated)
        SET(HAS_GENERATED_FILES TRUE)
        LIST(APPEND ${RTI_SUBMODULE_NAME}_GENERATED ${f})
    ENDIF()
ENDFOREACH()

IF (${HAS_GENERATED_FILES})
    SET_PROPERTY(GLOBAL APPEND PROPERTY ${RTI_MODULE_NAME}_${RTI_SUBMODULE_NAME}_HAS_GENERATE true)
    ADD_CUSTOM_TARGET(${RTI_MODULE_NAME}_${RTI_SUBMODULE_NAME}_GENERATE
                      DEPENDS ${${RTI_SUBMODULE_NAME}_GENERATED})
ENDIF()
