macro(SET_VARIABLE_FROM_ENV name default)
    if (DEFINED ENV{${name}} AND (NOT DEFINED ${name}))
        set(${name} $ENV{${name}})
    ELSEIF(NOT ${name})
        set(${name} ${default})
    endif()
endmacro()
