#===============================================================================
# FUNCTION ..: GET_MODULE_DIR
# SYNOPSIS ..: Determine if a module name should include the version or not
#===============================================================================
FUNCTION(GET_MODULE_DIR inname outname)
    IF (${RTIME_DEVTREE})
        SET(${outname} ${inname} PARENT_SCOPE)
    ELSE()
        STRING(REGEX REPLACE "([a-zA-Z_])\\.[0-9]+\\.[0-9]+" "\\1"
                        ${outname} ${inname})
    ENDIF()
ENDFUNCTION(GET_MODULE_DIR)

#===============================================================================
# FUNCTION ..: GET_MODULE_NAME
# SYNOPSIS ..: Determine if a module name should include the version or not
#===============================================================================
FUNCTION(GET_MODULE_NAME inname outname)
    STRING(REGEX REPLACE "([a-zA-Z_])\\.[0-9]+\\.[0-9]+" "\\1"
                        ${outname} ${inname})
    SET(${outname} ${${outname}} PARENT_SCOPE)
ENDFUNCTION(GET_MODULE_NAME)

#===============================================================================
# FUNCTION ..: INCLUDE_MODULE
# SYNOPSIS ..:
#===============================================================================
FUNCTION(INCLUDE_MODULE module_dir)
    SET(options NO_TEST ONLY_IF_EXISTS)
    SET(value_arg)
    SET(list_arg)
    cmake_parse_arguments(RTIME "${options}" "${value_arg}" "${list_arg}" ${ARGN})

    GET_MODULE_NAME(${module_dir} module_name)

    #SET(dds_c_EXPORT_NAME librti_me PARENT_SCOPE)

    IF (${RTIME_DEVTREE})
        SET(CURRENT_MODULE_DIR ${module_dir})
        SET(CURRENT_MODULE_TEST_DIR ${module_dir})
    ELSE()
        IF ("${module_dir}" MATCHES "test")
            SET(CURRENT_MODULE_DIR test/${module_name})
        ELSE()
            SET(CURRENT_MODULE_DIR src/${module_name})
        ENDIF()
        SET(CURRENT_MODULE_TEST_DIR test/${module_name})
    ENDIF()

    IF (RTIME_ONLY_IF_EXISTS
        AND NOT IS_DIRECTORY "${RTIME_SOURCE_ROOT}/${CURRENT_MODULE_DIR}")
        RETURN()
    ENDIF()

    # Calculate once if this module should build forge tests (available to all branches below)
    SET(forge_builds_module FALSE)
    IF (RTI_BUILD_FORGE_UNITTESTS)
        FORGE_TESTS_ENABLED_FOR_MODULE(forge_builds_module)
    ENDIF()

    IF ((${module_dir} MATCHES "test") OR NOT RTIME_PSL)
            IF (${module_name} MATCHES "test")
                INCLUDE(${CURRENT_MODULE_TEST_DIR}/CMakeLists.txt)
            ELSE()
                INCLUDE(${CURRENT_MODULE_DIR}/CMakeLists.txt)
            ENDIF()

        SET(${module_name}_SRC_C ${${module_name}_SRC_C} PARENT_SCOPE)
        SET(${module_name}_SRC_H ${${module_name}_SRC_H} PARENT_SCOPE)
        GET_PROPERTY(
            m_generate GLOBAL
            PROPERTY ${module_name}_HAS_GENERATE SET
        )

        IF (m_generate)
            SET(MODULE_GENERATE
                ${MODULE_GENERATE} ${module_name}_GENERATE PARENT_SCOPE
            )
        ENDIF()
        SET(MODULE_LIST ${MODULE_LIST} ${module_name} PARENT_SCOPE)
    ELSEIF (RTIME_PSL
            AND (EXISTS "${CURRENT_MODULE_DIR}/srcC/test"
                 OR EXISTS "${CURRENT_MODULE_DIR}/srcCxx/test")
             AND ${RTI_BUILD_UNITTESTS})
        SET(RTI_LINK_TESTS TRUE)
            IF (${module_name} MATCHES "test")
                INCLUDE(${CURRENT_MODULE_TEST_DIR}/CMakeLists.txt)
            ELSE()
                INCLUDE(${CURRENT_MODULE_DIR}/CMakeLists.txt)
            ENDIF()
        SET(RTI_LINK_TESTS FALSE)

        LINK_DIRECTORIES(
            ${RTIME_BUILD_ROOT}/cmake/${CMAKE_BUILD_TYPE}/${RTIME_PIL_ARCH}
        )

        # Check early if this module should build forge tests
        SET(build_forge FALSE)
        IF (RTI_BUILD_FORGE_UNITTESTS)
            FORGE_TESTS_ENABLED_FOR_MODULE(build_forge)
        ENDIF()

        # For Forge architectures, only create tester libs if module is in Forge list.
        # For non-Forge architectures, always create testers.
        SET(RTI_TESTER_TARGET_CREATED FALSE)
        IF (RTI_BUILD_FORGE_UNITTESTS AND NOT build_forge)
            # Forge architecture, but this module is not in Forge list - skip entirely
        ELSEIF (RTI_BUILD_UNITTESTS_AS_LIBS OR RTI_BUILD_FORGE_UNITTESTS)
            # Build as library (PIL/PSL or Forge)
            IF (EXISTS "${CURRENT_MODULE_DIR}/srcC/test")
                ADD_LIBRARY(${RTI_MODULE_TESTER_STATIC}
                                rti_me_psl.2.0/srcC/test/PSLTester.c
                )
            ELSE()
                ADD_LIBRARY(${RTI_MODULE_TESTER_STATIC}
                                rti_me_psl.2.0/srcC/test/PSLTester.cxx
                )
            ENDIF()
            SET(RTI_TESTER_TARGET_CREATED TRUE)
        ELSE()
            # Build as executable (non-PIL platforms)
            IF (EXISTS "${CURRENT_MODULE_DIR}/srcC/test")
                ADD_EXECUTABLE(${RTI_MODULE_TESTER_STATIC}
                                rti_me_psl.2.0/srcC/test/PSLTester.c
                )
            ELSE()
                ADD_EXECUTABLE(${RTI_MODULE_TESTER_STATIC}
                                rti_me_psl.2.0/srcC/test/PSLTester.cxx
                )
            ENDIF()
            SET(RTI_TESTER_TARGET_CREATED TRUE)
        ENDIF()

        # Run Forge build if enabled for this module
        IF (RTI_BUILD_FORGE_UNITTESTS AND build_forge)
            MODULE_BUILD_TESTS_WITH_FORGE()
        ENDIF()

        SET(libdeps)
        set(netiodeps)

        FOREACH(md ${RTI_MODULE_IMPORTS} ${RTI_MODULE_TEST_IMPORT})
            IF (${md} STREQUAL "dds_c")
                LIST(APPEND libdeps rti_me${RTI_LIB_SUFFIX_STATIC})
            ELSEIF (${md} STREQUAL "dds_cpp")
                LIST(APPEND libdeps rti_me_cpp${RTI_LIB_SUFFIX_STATIC})
            ELSEIF (DEFINED ${md}_EXPORT_NAME)
                IF (${${md}_EXPORT_NAME} MATCHES "^rti_me(.*)")
                    IF (${md} MATCHES "netio_.*")
                        LIST(APPEND netiodeps ${${md}_EXPORT_NAME}${RTI_LIB_SUFFIX_STATIC})
                    ELSE()
                        LIST(APPEND libdeps ${${md}_EXPORT_NAME}${RTI_LIB_SUFFIX_STATIC})
                    ENDIF()
                ELSE()
                    IF (${md} MATCHES "netio_.*")
                        LIST(APPEND netiodeps
                        rti_me_${${md}_EXPORT_NAME}${RTI_LIB_SUFFIX_STATIC})
                    ELSE()
                        LIST(APPEND libdeps
                        rti_me_${${md}_EXPORT_NAME}${RTI_LIB_SUFFIX_STATIC})
                    ENDIF()
                ENDIF()
            ELSE()
                IF (${md} MATCHES "netio_.*")
                    LIST(APPEND netiodeps rti_me_${md}${RTI_LIB_SUFFIX_STATIC})
                ELSE()
                    LIST(APPEND libdeps rti_me_${md}${RTI_LIB_SUFFIX_STATIC})
                ENDIF()
            ENDIF()
        ENDFOREACH()

        IF (${module_name} STREQUAL "dds_c")
            # Do not add dds_c. netiopsl already has a dependency on rti_me
            # and adding it here would result in a duplicate library being
            # added.
        ELSEIF (${module_name} STREQUAL "dds_cpp")
            SET(module_lib rti_me_cpp${RTI_LIB_SUFFIX_STATIC})
        ELSEIF (DEFINED ${module_name}_EXPORT_NAME)
            IF (${${module_name}_EXPORT_NAME} MATCHES "^rti_me(.*)")
                SET(module_lib ${${module_name}_EXPORT_NAME}${RTI_LIB_SUFFIX_STATIC})
            ELSE()
                SET(module_lib rti_me_${${module_name}_EXPORT_NAME}${RTI_LIB_SUFFIX_STATIC})
            ENDIF()
        ELSE()
            SET(module_lib rti_me_${module_name}${RTI_LIB_SUFFIX_STATIC})
        ENDIF()

        IF (RTI_TESTER_TARGET_CREATED)
            IF (EXISTS "${CURRENT_MODULE_DIR}/srcCxx/test")
                TARGET_LINK_LIBRARIES(${RTI_MODULE_TESTER_STATIC}
                                    ${module_name}Tester${RTI_LIB_SUFFIX_STATIC}
                                    ${module_lib}
                                    ${netiodeps}
                                    rti_me_netiopsl_cpp${RTI_LIB_SUFFIX_STATIC}
                                    ${libdeps}
                                    rti_me_ospsl${RTI_LIB_SUFFIX_STATIC}
                                    rti_me_test${RTI_LIB_SUFFIX_STATIC}
                                    ${PLATFORM_TEST_LIBS}
                )
            ELSE()
                TARGET_LINK_LIBRARIES(${RTI_MODULE_TESTER_STATIC}
                                    ${module_name}Tester${RTI_LIB_SUFFIX_STATIC}
                                    ${module_lib}
                                    ${netiodeps}
                                    rti_me_netiopsl${RTI_LIB_SUFFIX_STATIC}
                                    ${libdeps}
                                    rti_me_test${RTI_LIB_SUFFIX_STATIC}
                                    ${PLATFORM_TEST_LIBS}
                )
            ENDIF()

            ADD_CUSTOM_COMMAND(
                TARGET ${RTI_MODULE_TESTER_STATIC} POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E make_directory
                        ${RTI_MODULE_OUTPUT_SRC}/lib/${RTIME_TARGET_NAME}
                COMMAND ${CMAKE_COMMAND} -E copy
                        $<TARGET_FILE:${module_name}Tester${RTI_TESTER_SUFFIX}>
                        ${RTI_MODULE_OUTPUT_SRC}/lib/${RTIME_TARGET_NAME}/$<TARGET_FILE_NAME:${RTI_MODULE_TESTER_STATIC}>
            )

            rtime_add_unittest(LINK_ONLY)
        ENDIF()
    ELSE()
        IF(RTIME_PSL)
            IF (NOT RTIME_PIL_PATH)
                SET(RTIME_PIL_PATH "${RTIME_MODULE_EXPORT_ROOT}/lib/${RTIME_PIL_ARCH}")
            ENDIF()

            LINK_DIRECTORIES(
                ${RTIME_PIL_PATH}
            )
        ENDIF()
        INCLUDE_MODULE_API(${module_dir})
        set(${module_name}_MODULE_DIR ${${module_name}_MODULE_DIR})
        set(${module_name}_MODULE_DIR ${${module_name}_MODULE_DIR} PARENT_SCOPE)
        SET(${module_name}_EXPORT_NAME ${${module_name}_EXPORT_NAME})
        SET(${module_name}_EXPORT_NAME ${${module_name}_EXPORT_NAME} PARENT_SCOPE)
    ENDIF()
ENDFUNCTION(INCLUDE_MODULE)

#===============================================================================
# FUNCTION ..: INCLUDE_MODULE_API
# SYNOPSIS ..:
#===============================================================================
FUNCTION(INCLUDE_MODULE_API module_dir)
    SET(options ONLY_IF_EXISTS)
    SET(value_arg)
    SET(list_arg)
    cmake_parse_arguments(RTIME "${options}" "${value_arg}" "${list_arg}" ${ARGN})

    GET_MODULE_NAME(${module_dir} module_name)

    IF (${RTIME_DEVTREE})
        SET(CURRENT_MODULE_DIR ${module_dir})
    ELSE()
        SET(CURRENT_MODULE_DIR src/${module_name})
    ENDIF()

    IF(RTIME_ONLY_IF_EXISTS AND NOT IS_DIRECTORY "${CURRENT_MODULE_DIR}")
        RETURN()
    ENDIF()

    SET(RTIME_INCLUDE_API_ONLY TRUE)
    SET(RTIME_INCLUDE_DIR_ONLY TRUE)
    INCLUDE(${CURRENT_MODULE_DIR}/CMakeLists.txt)
    SET(RTIME_INCLUDE_API_ONLY FALSE)
    SET(RTIME_INCLUDE_DIR_ONLY FALSE)
ENDFUNCTION(INCLUDE_MODULE_API)

#===============================================================================
# FUNCTION ..: INCLUDE_PSL
# SYNOPSIS ..:
#===============================================================================
FUNCTION(INCLUDE_PSL module_dir)
    SET(options NETIO DDS NETIOCPP PSK)
    SET(value_arg OSAPI)
    SET(list_arg "")
    cmake_parse_arguments(RTIME_PSL "${options}" "${value_arg}" "${list_arg}" ${ARGN})

    IF (RTIME_PSL_OSAPI AND RTIME_FULL_SOURCE)
        INCLUDE_MODULE_API(${RTIME_PSL_OSAPI})
    endif()

    GET_MODULE_NAME(${module_dir} module_name)

    IF (${RTIME_DEVTREE})
        SET(CURRENT_MODULE_DIR ${module_dir})
    ELSE()
        SET(CURRENT_MODULE_DIR src/${module_name})
    ENDIF()

    IF (NOT RTIME_DEVTREE)
        IF (NOT RTIME_PIL_PATH)
            SET(RTIME_PIL_PATH "${RTIME_MODULE_EXPORT_ROOT}/lib/${RTIME_PIL_ARCH}")
        ENDIF()

        LINK_DIRECTORIES(
            ${RTIME_PIL_PATH}
        )
    ENDIF()

    SET(MODULE_LIST ${MODULE_LIST} ${module_name} PARENT_SCOPE)

    INCLUDE(${CURRENT_MODULE_DIR}/CMakeLists.txt)

ENDFUNCTION(INCLUDE_PSL)

# FUNCTION ..: RESOLVE_SHARED_COPY_COMMAND
# SYNOPSIS ..:
#===============================================================================
FUNCTION(RESOLVE_SHARED_COPY_COMMAND cmd src dst)
#    MESSAGE("resolving command: ${src} ${dst}")
#OR (${src} MATCHES "(.*)")
    IF (${src} MATCHES "rsync://(.*):(.*)")
        SET(${cmd} rsync -e "ssh -o StrictHostKeyChecking=no" -qaz ${CMAKE_MATCH_1}:${CMAKE_MATCH_2} ${dst} PARENT_SCOPE)
    ELSEIF ((${src} MATCHES "dir://(.*)") )
        SET(${cmd} ${CMAKE_COMMAND} -E copy ${CMAKE_MATCH_1} ${dst} PARENT_SCOPE)
    ELSEIF (${src} MATCHES "(.*)")
        SET(${cmd} ${CMAKE_COMMAND} -E copy ${CMAKE_MATCH_1} ${dst} PARENT_SCOPE)
    ENDIF()
#    MESSAGE("resolved command: ${${cmd}}")
ENDFUNCTION(RESOLVE_SHARED_COPY_COMMAND)

#===============================================================================
# FUNCTION ..: MICRO_ADD_IDL
# SYNOPSIS ..:
#===============================================================================
FUNCTION(MICRO_ADD_IDL IDL_GEN_C IDL_GEN_H IDL ROOT SRCDIR IDL_OUTPUT_DIR)

    SET(SRC_ROOT ${${ROOT}})
    SET(IDL_SOURCE ${${IDL}})
    SET(R_IDL_GEN_C)
    SET(R_IDL_GEN_H)

    LIST(LENGTH IDL_SOURCE l)

    IF (NOT ${l} GREATER 0)
        RETURN()
    ENDIF()

    FOREACH(idl ${IDL_SOURCE})
        SET(L_IDL_GEN_C)
        SET(L_IDL_GEN_H)
        set(NDDSGEN_OPT)

        if (${idl} MATCHES "(.*)::(.*)")
            set(idl ${CMAKE_MATCH_1})
            SET(NDDSGEN_OPT ${CMAKE_MATCH_2})
            STRING(REPLACE "\"" "" NDDSGEN_OPT "${NDDSGEN_OPT}")
            STRING(REPLACE " " ";" NDDSGEN_OPT "${NDDSGEN_OPT}")
        endif()

        STRING(REGEX REPLACE "\\.idl" "" basename ${idl})

        # If the idl file is located at a path that contains Cxx, and its name
        # does not contain _LangC, we will generate C++ code for that Idl type.
        IF (${SRCDIR} MATCHES "Cxx" AND NOT ${basename} MATCHES ".*LangC.*")
            SET(LANGUAGE "C++")
        ELSE()
            SET(LANGUAGE "C")
        ENDIF()

        IF (${LANGUAGE} STREQUAL "C++")
            SET(C_EXT "cxx")
            SET(H_EXT "h")
        ELSE()
            SET(C_EXT "c")
            SET(H_EXT "h")
        ENDIF()

        LIST(APPEND L_IDL_GEN_C ${IDL_OUTPUT_DIR}/${basename}.${C_EXT}
                                ${IDL_OUTPUT_DIR}/${basename}Plugin.${C_EXT}
                                ${IDL_OUTPUT_DIR}/${basename}Support.${C_EXT})

        LIST(APPEND L_IDL_GEN_H ${IDL_OUTPUT_DIR}/${basename}.${H_EXT}
                                ${IDL_OUTPUT_DIR}/${basename}Plugin.${H_EXT}
                                ${IDL_OUTPUT_DIR}/${basename}Support.${H_EXT})

        IF (${basename} MATCHES ".*Ns.*")
            LIST(APPEND NDDSGEN_OPT -namespace )
        ENDIF()

        IF (${basename} MATCHES ".*V1V2.*")
            LIST(APPEND NDDSGEN_OPT  -interpreted 1)
        ELSEIF(NOT "${LANGUAGE}" STREQUAL "C++")
            LIST(FIND NDDSGEN_OPT "-interpreted" index)
            IF (${index}  EQUAL -1)
                LIST(APPEND NDDSGEN_OPT  -interpreted 0)
            ENDIF()
        ENDIF()

        IF (CMAKE_GENERATOR MATCHES "Visual Studio")
            #Visual studio needs to run commands using 'call'
            #if there is more than one custom command
            SET(COMMAND_PREFIX call)
        ELSE()
            SET(COMMAND_PREFIX)
        ENDIF()

        IF (NOT TARGET_ONLY)
            ADD_CUSTOM_COMMAND(OUTPUT ${L_IDL_GEN_C} ${L_IDL_GEN_H}
                COMMAND ${COMMAND_PREFIX} ${CMAKE_COMMAND}
                            -E make_directory ${IDL_OUTPUT_DIR}
                COMMAND ${COMMAND_PREFIX} ${RTIDDSGEN} ${RTIDDSGEN_OPT}
                            -verbosity 1
                                            -micro -language ${LANGUAGE}
                                            -replace ${NDDSGEN_OPT} -I ${SRC_ROOT}
                                            -DNDDSGEN_EMPTY_VALUETYPE_NOT_SUPPORTED
                                            -d ${IDL_OUTPUT_DIR}
                                            ${SRC_ROOT}/${basename}.idl
                               VERBATIM
                               DEPENDS ${SRC_ROOT}/${basename}.idl rtiddsgen
        )
        ENDIF()

        SET_SOURCE_FILES_PROPERTIES(${L_IDL_GEN_C} ${L_IDL_GEN_H} PROPERTIES GENERATED 1)
        LIST(APPEND R_IDL_GEN_C ${L_IDL_GEN_C})
        LIST(APPEND R_IDL_GEN_H ${L_IDL_GEN_H})
    ENDFOREACH()
    SET(IDL_GEN_C ${R_IDL_GEN_C} PARENT_SCOPE)
    SET(IDL_GEN_H ${R_IDL_GEN_H} PARENT_SCOPE)
ENDFUNCTION()

        )
        SET_SOURCE_FILES_PROPERTIES(${L_TS_GEN} PROPERTIES GENERATED 1)
        LIST(APPEND R_TS_GEN ${L_TS_GEN})
    ENDFOREACH()
    SET(${QUACK_TS_GEN} ${R_TS_GEN} PARENT_SCOPE)
ENDFUNCTION()

#===============================================================================
# FUNCTION ..: MICRO_ADD_XML_APPGEN
# SYNOPSIS ..:
#===============================================================================
FUNCTION(MICRO_ADD_XML_APPGEN XML_APPGEN_GEN_C XML_APPGEN_GEN_H XML_APPGEN ROOT SRCDIR XML_APPGEN_OUTPUT_DIR)

    SET(SRC_ROOT ${${ROOT}})
    SET(XML_APPGEN_SOURCE ${${XML_APPGEN}})
    SET(R_XML_APPGEN_GEN_C)
    SET(R_XML_APPGEN_GEN_H)

    IF (${SRCDIR} MATCHES "Cxx")
        SET(LANGUAGE "C++")
    ELSE()
        SET(LANGUAGE "C")
    ENDIF()

    LIST(LENGTH XML_APPGEN_SOURCE l)

    IF (NOT ${l} GREATER 0)
        RETURN()
    ENDIF()

    FOREACH(xml ${XML_APPGEN_SOURCE})
        SET(L_XML_APPGEN_GEN_C)
        SET(L_XML_APPGEN_GEN_H)

        # Parse filename and optional custom options (format: filename.xml::opt1;opt2;opt3)
        SET(xml_file "${xml}")
        SET(custom_options "")
        IF("${xml}" MATCHES "^(.*)::(.*)$")
            SET(xml_file "${CMAKE_MATCH_1}")
            SET(custom_options "${CMAKE_MATCH_2}")
        ENDIF()

        STRING(REGEX REPLACE "\\.xml" "" basename ${xml_file})

        IF (${LANGUAGE} STREQUAL "C")
            LIST(APPEND L_XML_APPGEN_GEN_C ${XML_APPGEN_OUTPUT_DIR}/${basename}Appgen.c)
            LIST(APPEND L_XML_APPGEN_GEN_H ${XML_APPGEN_OUTPUT_DIR}/${basename}Appgen.h)
        ELSE()
            LIST(APPEND L_XML_APPGEN_GEN_C ${XML_APPGEN_OUTPUT_DIR}/${basename}Appgen.c
                                           ${XML_APPGEN_OUTPUT_DIR}/${basename}AppgenPlugin.cxx)
            LIST(APPEND L_XML_APPGEN_GEN_H ${XML_APPGEN_OUTPUT_DIR}/${basename}Appgen.h
                                           ${XML_APPGEN_OUTPUT_DIR}/${basename}AppgenPlugin.h)
        ENDIF()

        # Use custom options if provided, otherwise use default
        IF(custom_options)
            STRING(REPLACE "," ";" mag_options "${custom_options}")
        ELSE()
            SET(mag_options "-dontUpdateResourceLimits")
        ENDIF()

        IF (NOT TARGET_ONLY)
            ADD_CUSTOM_COMMAND(OUTPUT ${L_XML_APPGEN_GEN_C} ${L_XML_APPGEN_GEN_H}
                               COMMAND ${CMAKE_COMMAND} -E make_directory ${XML_APPGEN_OUTPUT_DIR}
                               COMMAND ${RTIDDSMAG} ${RTIDDSMAG_OPT} -language ${LANGUAGE}
                                                 -replace ${mag_options}
                                                 -d ${XML_APPGEN_OUTPUT_DIR} ${SRC_ROOT}/${basename}.xml
                                                 -referencedFile ${SRC_ROOT}/${basename}Qos.xml
                               DEPENDS ${SRC_ROOT}/${basename}.xml rtiddsmag
            )
        ENDIF()

        SET_SOURCE_FILES_PROPERTIES(${L_XML_APPGEN_GEN_C} ${L_XML_APPGEN_GEN_H} PROPERTIES GENERATED 1)
        LIST(APPEND R_XML_APPGEN_GEN_C ${L_XML_APPGEN_GEN_C})
        LIST(APPEND R_XML_APPGEN_GEN_H ${L_XML_APPGEN_GEN_H})
    ENDFOREACH()
    SET(XML_APPGEN_GEN_C ${R_XML_APPGEN_GEN_C} PARENT_SCOPE)
    SET(XML_APPGEN_GEN_H ${R_XML_APPGEN_GEN_H} PARENT_SCOPE)
ENDFUNCTION()

function(process_resource_bundle_file path outpath file)
    ADD_CUSTOM_COMMAND(OUTPUT ${outpath}/${file}
          COMMAND ${CMAKE_COMMAND} -E copy ${path}/${file} ${outpath}/${file}
          DEPENDS ${path}/${file}
   )
endfunction(process_resource_bundle_file)

#===============================================================================
# FUNCTION ..: process_resource_bundle_dir
# SYNOPSIS ..:
#===============================================================================

function(process_resource_bundle_dir path outpath dir)
    include(${path}/CMakeLists.txt)

    set(local_resource_dir ${RTI_TEST_BUNDLE_RESOURCE_DIR})
    set(local_resource_file ${RTI_TEST_BUNDLE_RESOURCE_FILE})

    if (NOT EXISTS ${path})
        message(FATAL_ERROR "${path} does not exist")
    endif()

    set(local_return_value)
    foreach (rd ${local_resource_dir})
        IF(${rd} MATCHES "(.*)::(.*)")
            SET(rd "${CMAKE_MATCH_1}")
            SET(sub "/${CMAKE_MATCH_2}")
        ELSE()
            SET(sub "/${rd}")
        ENDIF()
        process_resource_bundle_dir(${path}/${rd} ${outpath}${sub} ${rd})
        LIST(APPEND local_return_value ${return_value})
    endforeach(rd)

    foreach (f ${local_resource_file})
        process_resource_bundle_file(${path} ${outpath} ${f})
        list(APPEND local_return_value  "${outpath}/${f}")
    endforeach(f)

    set(return_value ${local_return_value} PARENT_SCOPE)
endfunction(process_resource_bundle_dir)

#===============================================================================
# FUNCTION ..: process_resource_bundle_module
# SYNOPSIS ..:
#===============================================================================

function(process_resource_bundle_module module  output_root type resdir)
    SET(TEST_BUNDLE TRUE)
    if (NOT EXISTS ${module})
        message(FATAL_ERROR "${module} does not exists")
    endif()

    if (NOT EXISTS ${module}/CMakeLists.txt)
        return()
    endif()

    if (NOT "${resdir}" STREQUAL "")
        SET(rm resource/${resdir})
    ELSE()
        SET(rm resource)
    ENDIF()

    include(${module}/CMakeLists.txt)

    set(local_resource_dir ${RTI_TEST_BUNDLE_RESOURCE_DIR})
    set(local_resource_file ${RTI_TEST_BUNDLE_RESOURCE_FILE})

    # Make sure these are reset just in case they are not used in a
    # CMakeLists.txt file
    SET(RTI_TEST_BUNDLE_RESOURCE_FILE)
    SET(RTI_TEST_BUNDLE_RESOURCE_DIR)

    set(local_return_value)
    foreach (rd ${local_resource_dir})
        IF(${rd} MATCHES "(.*)::(.*)")
            SET(rd "${CMAKE_MATCH_1}")
            SET(sub "/${CMAKE_MATCH_2}")
        ELSE()
            SET(sub "/${rd}")
        ENDIF()
        process_resource_bundle_dir(${module}/${rd} ${output_root}/${type}/${rm}${sub} ${rd})
        list(APPEND local_return_value ${return_value})
    endforeach(rd)

    foreach (f ${local_resource_file})
        process_resource_bundle_file(${module} ${rm} ${f})
        list(APPEND local_return_value "${rm}/${f}")
    endforeach(f)

    set(return_value ${local_return_value} PARENT_SCOPE)
    SET(TEST_BUNDLE FALSE)
endfunction(process_resource_bundle_module)

# USAGE .....: GENERATE_TEST_MANIFEST(
#                  TARGET target_name
#                  OUTPUT_DIR directory
#                  TEST_EXECUTABLE executable_name
#                  ENTRY_POINT entry_point_name
#                  [FULL_TEST_NAME override_name]
#              )
#===============================================================================
FUNCTION(GENERATE_TEST_MANIFEST)
    SET(options)
    SET(oneValueArgs TARGET OUTPUT_DIR TEST_EXECUTABLE ENTRY_POINT FULL_TEST_NAME)
    SET(multiValueArgs)
    CMAKE_PARSE_ARGUMENTS(GEN "${options}" "${oneValueArgs}" "${multiValueArgs}" ${ARGN})

    IF(GEN_FULL_TEST_NAME)
        SET(MANIFEST_FULL_NAME "${GEN_FULL_TEST_NAME}")
    ELSE()
        SET(MANIFEST_FULL_NAME "${RTI_MODULE_BUNDLE_PREFIX}::${RTI_MODULE_NAME}::${RTI_MODULE_TESTER_BUNDLE}")
    ENDIF()

    STRING(TIMESTAMP RTIME_BUNDLE_CREATION_TIMESTAMP)

    ADD_CUSTOM_COMMAND(TARGET ${GEN_TARGET} POST_BUILD
        COMMAND ${CMAKE_COMMAND}
            -DRTIME_SOURCE_ROOT=${RTIME_SOURCE_ROOT}

            -DCOMMIT_ID=${RTI_COMMIT_ID}
            -DARCHITECTURE=${RTIME_TARGET_NAME}
            -DCREATE_TIMESTAMP=${RTIME_BUNDLE_CREATION_TIMESTAMP}
            -DRTI_BUNDLE_TYPE=${RTI_BUNDLE_TYPE}
            -DFULL_TEST_NAME="${MANIFEST_FULL_NAME}"
            -DTEST_EXECUTABLE=${GEN_TEST_EXECUTABLE}
            -DENTRY_POINT=${GEN_ENTRY_POINT}
            -DINPUT_FILE=${RTIME_RESOURCE_ROOT}/cmake/cmake_test_manifest.cmake
            -DOUTPUT_FILE=${GEN_OUTPUT_DIR}/MANIFEST
            -P ${RTIME_RESOURCE_ROOT}/cmake/cmake_generate_test_bundle.cmake
        COMMENT "Generating test manifest for ${RTI_MODULE_NAME}"
    )
ENDFUNCTION(GENERATE_TEST_MANIFEST)

function(rtime_add_unittest)
    if (NOT ${RTI_BUILD_UNITTESTS} OR NOT RTIME_UTEST_QUACK_ENABLE)
        return()
    endif()
    SET(options LINK_ONLY)
    SET(value_arg)
    SET(list_arg)
    cmake_parse_arguments(RTIME "${options}" "${value_arg}" "${list_arg}" ${ARGN})

    IF (RTIME_UTEST_QUACK_ENABLE AND (NOT ${RTI_MODULE_NAME} MATCHES "rti_me_psl"))
        process_resource_bundle_module(${RTIME_SOURCE_ROOT}/${CURRENT_MODULE_DIR}
                                       ${RTI_TARGET_BUNDLE_ROOT}
                                       "static/${RTI_MODULE_TESTER_BUNDLE}"
                                       ${RTI_MODULE_NAME}
        )

        ADD_CUSTOM_TARGET(${RTI_MODULE_NAME}_resources
                          DEPENDS ${return_value}
        )
    ENDIF()

    FOREACH(srcdir "" /srcC /srcCxx)
        FOREACH(sm ${RTI_SUBMODULES})
            UNSET(RTI_SUBMODULE_TEST_QUACK_TS)
            UNSET(RTI_SUBMODULE_TEST_QUACK_TI)
            UNSET(RTI_SUBMODULE_TEST_QUACK_TS_GEN)
            UNSET(RTI_SUBMODULE_QUACK_TS)
            UNSET(RTI_SUBMODULE_QUACK_TI)
            UNSET(RTI_SUBMODULE_QUACK_TS_GEN)

            IF (EXISTS ${RTI_MODULE_TESTROOT}${srcdir}/${sm}/test/CMakeLists.txt)
                SET(SRC_ROOT "${RTI_MODULE_SRCROOT}${srcdir}/${sm}/test")
                INCLUDE(${RTI_MODULE_TESTROOT}${srcdir}/${sm}/test/CMakeLists.txt)

                IF (DEFINED RTI_SUBMODULE_TEST_QUACK_TS)
                    MICRO_ADD_QUACK_TS(RTI_SUBMODULE_TEST_QUACK_TS_GEN
                                       RTI_SUBMODULE_TEST_QUACK_TS
                                       SRC_ROOT ${srcdir}
                                       "${RTIME_QUACK_TEST_ROOT}/${RTI_MODULE_NAME}/${sm}/test")
                    LIST(APPEND ${RTI_MODULE_NAME}_TEST_QUACK_TS ${RTI_SUBMODULE_TEST_QUACK_TS_GEN})
                ENDIF()

                IF (DEFINED RTI_SUBMODULE_TEST_QUACK_TI)
                    MICRO_ADD_QUACK_TI(RTI_SUBMODULE_TEST_QUACK_TI_GEN
                                       RTI_SUBMODULE_TEST_QUACK_TI
                                       SRC_ROOT ${srcdir}
                                       "${RTIME_QUACK_TEST_ROOT}/${RTI_MODULE_NAME}/${sm}/test")
                    LIST(APPEND ${RTI_MODULE_NAME}_TEST_QUACK_TI ${RTI_SUBMODULE_TEST_QUACK_TI_GEN})
                ENDIF()
            ENDIF()
        ENDFOREACH()
    ENDFOREACH()

    IF (RTIME_UTEST_QUACK_ENABLE AND NOT RTIME_PIL)
        ADD_DEPENDENCIES(${RTI_MODULE_TESTER_STATIC}
                         bundle_common
                         ${RTI_MODULE_NAME}_resources
        )

        IF (DEFINED ${RTI_MODULE_NAME}_QUACK_TS)
            ADD_CUSTOM_TARGET(${RTI_MODULE_NAME}_QUACK_TS_TARGET
                            DEPENDS ${${RTI_MODULE_NAME}_QUACK_TS}
            )

            ADD_DEPENDENCIES(${RTI_MODULE_TESTER_STATIC}
                            ${RTI_MODULE_NAME}_QUACK_TS_TARGET
            )
        ENDIF()

        IF (DEFINED ${RTI_MODULE_NAME}_QUACK_TI)
            ADD_CUSTOM_TARGET(${RTI_MODULE_NAME}_QUACK_TI_TARGET
                              DEPENDS ${${RTI_MODULE_NAME}_QUACK_TI}
            )

            ADD_DEPENDENCIES(${RTI_MODULE_TESTER_STATIC}
                             ${RTI_MODULE_NAME}_QUACK_TI_TARGET
            )
        ENDIF()

        IF (DEFINED ${RTI_MODULE_NAME}_TEST_QUACK_TS)
            ADD_CUSTOM_TARGET(${RTI_MODULE_NAME}_TEST_QUACK_TS_TARGET
                            DEPENDS ${${RTI_MODULE_NAME}_TEST_QUACK_TS}
            )

            ADD_DEPENDENCIES(${RTI_MODULE_TESTER_STATIC}
                            ${RTI_MODULE_NAME}_TEST_QUACK_TS_TARGET
            )
        ENDIF()

        IF (DEFINED ${RTI_MODULE_NAME}_TEST_QUACK_TI)
            ADD_CUSTOM_TARGET(${RTI_MODULE_NAME}_TEST_QUACK_TI_TARGET
                              DEPENDS ${${RTI_MODULE_NAME}_TEST_QUACK_TI}
            )

            ADD_DEPENDENCIES(${RTI_MODULE_TESTER_STATIC}
                             ${RTI_MODULE_NAME}_TEST_QUACK_TI_TARGET
            )
        ENDIF()
    ENDIF()

    # Generate MANIFEST using common function
    GENERATE_TEST_MANIFEST(
        TARGET ${RTI_MODULE_TESTER_STATIC}
        OUTPUT_DIR ${RTI_TARGET_BUNDLE_ROOT}/static/${RTI_MODULE_TESTER_BUNDLE}
        TEST_EXECUTABLE $<TARGET_FILE_NAME:${RTI_MODULE_TESTER_STATIC}>
        ENTRY_POINT main
    )

    # Copy test executable to bundle directory
    ADD_CUSTOM_COMMAND(TARGET ${RTI_MODULE_TESTER_STATIC} POST_BUILD
       COMMAND ${CMAKE_COMMAND} -E copy
               $<TARGET_FILE:${RTI_MODULE_TESTER_STATIC}>
               ${RTI_TARGET_BUNDLE_ROOT}/static/${RTI_MODULE_TESTER_BUNDLE}/lib/${RTIME_TARGET_NAME}/$<TARGET_FILE_NAME:${RTI_MODULE_TESTER_STATIC}>
    )


    ADD_CUSTOM_COMMAND(TARGET ${RTI_MODULE_TESTER_STATIC} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E echo "Create bundle \
            ${RTI_TARGET_BUNDLE_ROOT}/static/${RTI_MODULE_TESTER_BUNDLE}.tgz \
            in ${RTI_TARGET_BUNDLE_ROOT}/static/${RTI_MODULE_TESTER_BUNDLE}"

            COMMAND ${CMAKE_COMMAND} -E copy_directory
                ${RTI_TARGET_BUNDLE_ROOT}/static/resource
                ${RTI_TARGET_BUNDLE_ROOT}/static/${RTI_MODULE_TESTER_BUNDLE}/resource

            COMMAND ${CMAKE_COMMAND} -E tar cz
                ${RTI_TARGET_BUNDLE_ROOT}/static/${RTI_MODULE_TESTER_BUNDLE}.tgz -- .

            WORKING_DIRECTORY ${RTI_TARGET_BUNDLE_ROOT}/static/${RTI_MODULE_TESTER_BUNDLE}
    )

    if (HAS_RM)
        ADD_CUSTOM_COMMAND(TARGET ${RTI_MODULE_TESTER_STATIC} POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E rm -r -- ${RTI_TARGET_BUNDLE_ROOT}/static/${RTI_MODULE_TESTER_BUNDLE}
        )
    else()
        ADD_CUSTOM_COMMAND(TARGET ${RTI_MODULE_TESTER_STATIC} POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E remove_directory ${RTI_TARGET_BUNDLE_ROOT}/static/${RTI_MODULE_TESTER_BUNDLE}
        )
    endif()
endfunction(rtime_add_unittest)

#===============================================================================
# FUNCTION ..: rtime_add_shared_library
# SYNOPSIS ..:
#===============================================================================

function(rtime_add_shared_library)
    set(options INSTALL_LIBRARY)
    set(value_arg LIBRARY_NAME)
    set(list_arg SOURCE HEADER DEPENDS)
    cmake_parse_arguments(RTIME "${options}" "${value_arg}" "${list_arg}" ${ARGN})

    IF (RTIME_NO_SHARED_LIB)
        return()
    endif()

    SET(RTI_EXPORT_SUBMODULE_OUTPUT_LIB ${RTIME_MODULE_EXPORT_ROOT}/lib/${RTIME_TARGET_NAME})

    SET(RTI_SUBMODULE_LIBNAME_SHARED
        ${RTI_LIB_PREFIX}_${RTIME_LIBRARY_NAME}${RTI_LIB_SUFFIX_DYNAMIC}
    )

    ADD_LIBRARY(${RTI_SUBMODULE_LIBNAME_SHARED} SHARED
                ${RTIME_SOURCE} ${RTIME_HEADER}
    )

    IF (RTIME_INSTALL_LIBRARY AND RTIME_FULL_SOURCE AND NOT RTIME_DEVTREE)
        IF (NOT RTIME_INSTALL_DIR)
            SET(RTI_INSTALL_DIR "lib")
        ENDIF()
        INSTALL(TARGETS
                    ${RTI_SUBMODULE_LIBNAME_SHARED}
                RUNTIME
                    DESTINATION "${RTI_INSTALL_DIR}"
                LIBRARY
                    DESTINATION "${RTI_INSTALL_DIR}"
                ARCHIVE
                    DESTINATION "${RTI_INSTALL_DIR}"
        )
    ENDIF()

    FOREACH(mdin ${RTI_MODULE_DEPENDS} ${RTI_MODULE_IMPORTS})
        IF (${mdin} MATCHES "(.*)::(.*)")
            SET(mdin "${CMAKE_MATCH_1}")
        ENDIF()
        SET_PROPERTY(TARGET ${RTI_SUBMODULE_LIBNAME_SHARED} APPEND
                    PROPERTY COMPILE_DEFINITIONS RTI_${mdin}_DLL_VARIABLE
        )

        SET_TARGET_PROPERTIES(${RTI_SUBMODULE_LIBNAME_SHARED}
                    PROPERTIES CMAKE_SKIP_RPATH TRUE
        )
    ENDFOREACH()

    FOREACH(md ${RTI_MODULE_IMPORTS} ${RTI_LIBRARY_DEPENDS})
        IF (${md} MATCHES "(.*)::(.*)")
            set(md "${CMAKE_MATCH_2}")
        ENDIF()
        add_exported_header_dependency(
                DEPEDENCY ${md}
                DEPENDENTS ${RTI_SUBMODULE_LIBNAME_SHARED}
        )
        IF (DEFINED ${md}_EXPORT_NAME)
            LIST(APPEND LIB_DEPS_DYNAMIC
                ${${md}_EXPORT_NAME}${RTI_LIB_SUFFIX_DYNAMIC}
            )
        ELSEIF(${md} MATCHES "lib.*")
            STRING(REGEX REPLACE "lib(.*)" "\\1" libname ${md})
            LIST(APPEND LIB_DEPS_DYNAMIC ${libname}${RTI_LIB_SUFFIX_DYNAMIC})
        ELSE()
            LIST(APPEND LIB_DEPS_DYNAMIC
                ${RTI_LIB_PREFIX}_${md}${RTI_LIB_SUFFIX_DYNAMIC}
            )
        ENDIF()
    ENDFOREACH()

    TARGET_INCLUDE_DIRECTORIES(${RTI_SUBMODULE_LIBNAME_SHARED}
                                 PRIVATE ${RTI_MODULE_INCLUDE_DIRS})

    TARGET_LINK_LIBRARIES(${RTI_SUBMODULE_LIBNAME_SHARED}
                          PUBLIC ${LIB_DEPS_DYNAMIC} ${PLATFORM_LIBS})

    # For windows we need to copy .exp,*.lib., *.pdb and *.dll
    IF (NOT RTIME_PIL)
        IF (${CMAKE_SHARED_LIBRARY_SUFFIX} STREQUAL ".dll")
            SET(libsuffix "dll" "lib" "exp")
            IF (${RTI_DEBUG_BUILD})
                LIST(APPEND libsuffix "pdb")
            ENDIF()
            FOREACH(f ${libsuffix})
                ADD_CUSTOM_COMMAND(TARGET ${RTI_SUBMODULE_LIBNAME_SHARED} POST_BUILD
                    COMMAND ${CMAKE_COMMAND} -E copy
                    $<TARGET_LINKER_FILE_DIR:${RTI_SUBMODULE_LIBNAME_SHARED}>/${CMAKE_SHARED_LIBRARY_PREFIX}${RTI_SUBMODULE_LIBNAME_SHARED}.${f}
                    ${RTI_EXPORT_SUBMODULE_OUTPUT_LIB}/${CMAKE_SHARED_LIBRARY_PREFIX}${RTI_SUBMODULE_LIBNAME_SHARED}.${f}
                )
            ENDFOREACH()
        ELSE()
            ADD_CUSTOM_COMMAND(TARGET ${RTI_SUBMODULE_LIBNAME_SHARED} POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E copy
                $<TARGET_LINKER_FILE:${RTI_SUBMODULE_LIBNAME_SHARED}>
                ${RTI_EXPORT_SUBMODULE_OUTPUT_LIB}/${CMAKE_SHARED_LIBRARY_PREFIX}${RTI_SUBMODULE_LIBNAME_SHARED}${CMAKE_SHARED_LIBRARY_SUFFIX}
            )
        ENDIF()
    ENDIF()

    FOREACH(mdin ${RTI_MODULE_IMPORTS} ${RTI_MODULE_DEPENDS})
        STRING(REGEX REPLACE "lib(.*)" "\\1" md ${mdin})
        IF (${mdin} MATCHES "(.*)::(.*)")
            SET(mdindll "${CMAKE_MATCH_1}")
            SET_PROPERTY(TARGET ${RTI_SUBMODULE_LIBNAME_SHARED} APPEND
                PROPERTY COMPILE_DEFINITIONS RTI_${CMAKE_MATCH_2}_DLL_VARIABLE)
        ELSE()
            SET(mdindll "${mdin}")
        ENDIF()
        SET_PROPERTY(TARGET ${RTI_SUBMODULE_LIBNAME_SHARED} APPEND
            PROPERTY COMPILE_DEFINITIONS RTI_${mdindll}_DLL_VARIABLE)
        IF (DEFINED ${mdin}_OBJECT_DEPENDS)
            FOREACH(mdin2 ${${mdin}_OBJECT_DEPENDS})
                IF (${mdin2} MATCHES "(.*)::(.*)")
                    SET(mdindll "${CMAKE_MATCH_1}")
                    SET_PROPERTY(TARGET ${RTI_SUBMODULE_LIBNAME_SHARED} APPEND
                        PROPERTY COMPILE_DEFINITIONS RTI_${CMAKE_MATCH_2}_DLL_VARIABLE)
                ELSE()
                    SET(mdindll "${mdin}")
                ENDIF()
                SET_PROPERTY(TARGET ${RTI_SUBMODULE_LIBNAME_SHARED} APPEND
                    PROPERTY COMPILE_DEFINITIONS RTI_${mdindll}_DLL_VARIABLE)
            ENDFOREACH()
        ENDIF()
    ENDFOREACH()

    FOREACH(mdin ${RTI_MODULE_EXPORTS})
        STRING(REGEX REPLACE "lib(.*)" "\\1" md ${mdin})
        IF (${md} MATCHES "(.*)::(.*)")
            SET(md "${CMAKE_MATCH_1}")
        ENDIF()
        SET_PROPERTY(TARGET
                        ${RTI_SUBMODULE_LIBNAME_SHARED}
                    APPEND PROPERTY COMPILE_DEFINITIONS
                        RTI_${md}_DLL_EXPORT
                        RTI_${md}_DLL_VARIABLE
        )
    ENDFOREACH()

    SET_TARGET_PROPERTIES(${RTI_SUBMODULE_LIBNAME_SHARED}
                    PROPERTIES
                    MACOSX_RPATH TRUE
    )

    SET_PROPERTY(TARGET
                    ${RTI_SUBMODULE_LIBNAME_SHARED}
                APPEND PROPERTY COMPILE_DEFINITIONS
                        RTI_${RTI_MODULE_NAME}_DLL_EXPORT
                        RTI_${RTI_SUBMODULE_NAME}_DLL_EXPORT
                        RTI_${RTI_MODULE_NAME}_DLL_VARIABLE
                        RTI_${RTI_SUBMODULE_NAME}_DLL_VARIABLE
    )

    SET_PROPERTY(TARGET ${RTI_SUBMODULE_LIBNAME_SHARED}  APPEND
                    PROPERTY COMPILE_DEFINITIONS
                    RTIME_TARGET_NAME=\"${RTIME_TARGET_NAME}\"
    )

    FOREACH(dep ${RTIME_DEPENDS})
        TARGET_INCLUDE_DIRECTORIES(
            ${RTI_SUBMODULE_LIBNAME_SHARED}
            PRIVATE "${RTIME_${dep}_INCLUDE_DIRECTORIES}"
        )

        TARGET_LINK_LIBRARIES(
            ${RTI_SUBMODULE_LIBNAME_SHARED}
            PUBLIC "${RTIME_${dep}_LINK_LIBRARIES}"
        )
        ADD_DEPENDENCIES(
            ${RTI_SUBMODULE_LIBNAME_SHARED}
            "${dep}"
        )
    ENDFOREACH()
endfunction()

            ENDFOREACH()
        ELSE()
            ADD_CUSTOM_COMMAND(TARGET ${RTI_SUBMODULE_LIBNAME_STATIC} POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E make_directory
                        ${RTI_EXPORT_SUBMODULE_OUTPUT_LIB}
                COMMAND ${CMAKE_COMMAND} -E copy
                $<TARGET_FILE:${RTI_SUBMODULE_LIBNAME_STATIC}>
                            ${RTI_EXPORT_SUBMODULE_OUTPUT_LIB}/${CMAKE_STATIC_LIBRARY_PREFIX}${RTI_SUBMODULE_LIBNAME_STATIC}${CMAKE_STATIC_LIBRARY_SUFFIX}
            )
        ENDIF()
    ENDIF()

    FOREACH(dep ${RTIME_DEPENDS})
        TARGET_INCLUDE_DIRECTORIES(
            ${RTI_SUBMODULE_LIBNAME_STATIC}
            PRIVATE "${RTIME_${dep}_INCLUDE_DIRECTORIES}"
        )

        TARGET_LINK_LIBRARIES(
            ${RTI_SUBMODULE_LIBNAME_STATIC}
            "${RTIME_${dep}_LINK_LIBRARIES}"
        )
        ADD_DEPENDENCIES(
            ${RTI_SUBMODULE_LIBNAME_STATIC}
            "${dep}"
        )
    ENDFOREACH()

endfunction()

#===============================================================================
# FUNCTION ..: rtime_add_library_src
# SYNOPSIS ..:
#===============================================================================

function(rtime_add_library_src folder lang library_name)
    set(out_src)
    set(out_hdr)
    set(out_cmakelist)

    if (RTIME_DEVTREE)
        set(src_root "${RTI_MODULE_SRCROOT}/src${lang}/${folder}")
    else()
        set(src_root "${RTI_MODULE_SRCROOT}/${folder}")
    endif()
    include("${src_root}/CMakeLists.txt")
    list(APPEND out_cmakelist "${src_root}/CMakeLists.txt")

    foreach(f ${RTI_PSL_CFLAGS})
        IF (NOT RTI_CMAKE_SOURCE)
            ADD_DEFINITIONS("${f}")
        ENDIF()
    endforeach()

    foreach(f ${RTI_PSL_SRC})
        list(APPEND out_src "${src_root}/${f}")
    endforeach()

    foreach(f ${RTI_PSL_HDR})
        list(APPEND out_hdr "${src_root}/${f}")
    endforeach()

    foreach(f ${RTI_PSL_SUBDIR})
        rtime_add_library_src("${src_root}/${f}" out_src out_hdr library_name)
        list(APPEND out_src ${RET_SOURCE})
        list(APPEND out_hdr ${RET_HEADER})
    endforeach()

    set(RET_SOURCE ${out_src} PARENT_SCOPE)
    set(RET_HEADER ${out_hdr} PARENT_SCOPE)
    set(RET_CMAKELIST ${out_cmakelist} PARENT_SCOPE)
endfunction()

function(rtime_add_psl_library)
    SET(SOURCE)
    SET(HEADER)
    SET(CMAKELIST)
    set(options)
    set(value_arg SRCROOT OUTPUT_SRC OUTPUT_HDR OUTPUT_CMAKELIST LIBRARY_NAME LANG)
    set(list_arg SUBMODULES DEPENDS)
    cmake_parse_arguments(RTIME "${options}" "${value_arg}" "${list_arg}" ${ARGN})

    FOREACH(m ${RTIME_SUBMODULES})
        rtime_add_library_src("${RTIME_SRCROOT}/${m}" "${RTIME_LANG}" "${RTIME_LIBRARY_NAME}")
        LIST(APPEND SOURCE ${RET_SOURCE})
        LIST(APPEND HEADER ${RET_HEADER})
        LIST(APPEND CMAKELIST ${RET_CMAKELIST})
    ENDFOREACH()

    set(${RTIME_OUTPUT_SRC} ${SOURCE} PARENT_SCOPE)
    set(${RTIME_OUTPUT_HDR} ${HEADER} PARENT_SCOPE)
    set(${RTIME_OUTPUT_CMAKELIST} ${CMAKELIST} PARENT_SCOPE)

    if (RTI_CMAKE_SOURCE OR TEST_BUNDLE)
        SET(RTI_SUBMODULE_SRC ${SOURCE} PARENT_SCOPE)
        SET(RTI_SUBMODULE_HDR ${HEADER} PARENT_SCOPE)
        return()
    endif()

    IF (RTI_OBJECT_MODULE)
        return()
    ENDIF()

    IF (NOT RTIME_PIL)
        SET(options INSTALL_LIBRARY)
    ENDIF()

    SET(ARGS LIBRARY_NAME ${RTIME_LIBRARY_NAME}
             SOURCE ${SOURCE}
             HEADER ${HEADER}
    )
    IF(DEFINED RTIME_DEPENDS)
        LIST(APPEND ARGS DEPENDS ${RTIME_DEPENDS})
    ENDIF()
    LIST(APPEND ARGS ${options})

    rtime_add_static_library(${ARGS})
    rtime_add_shared_library(${ARGS})

    SET(SM_HAS_GENERATED_FILES FALSE)
    SET(${RTI_SUBMODULE_NAME}_GENERATED)

    FOREACH(f ${${RTI_SUBMODULE_NAME}_SRC_C} ${${RTI_SUBMODULE_NAME}_SRC_H})
        GET_PROPERTY(rti_generated
                     SOURCE ${f}
                     PROPERTY RTI_GENERATED
                     SET)
        IF (rti_generated)
            SET(HAS_GENERATED_FILES TRUE)
            LIST(APPEND ${RTI_SUBMODULE_NAME}_GENERATED ${f})
        ENDIF()
    ENDFOREACH()

    IF (${HAS_GENERATED_FILES})
        SET_PROPERTY(GLOBAL APPEND PROPERTY ${RTI_MODULE_NAME}_${RTI_SUBMODULE_NAME}_HAS_GENERATE true)
        ADD_CUSTOM_TARGET(${RTI_MODULE_NAME}_${RTI_SUBMODULE_NAME}_GENERATE
                          DEPENDS ${${RTI_SUBMODULE_NAME}_GENERATED})
    ENDIF()

endfunction()

# FUNCTION ....: rtime_add_ifdoc
# PARAMETERS ..:
# SYNOPSIS ....:
#===============================================================================
FUNCTION(rtime_add_ifdoc)
    SET(IFDOC)
    SET(options)
    SET(value_arg IFDOC_SRCDIR IFDOC_DSTDIR)
    SET(list_arg IFDOC_SRC IFDOC_LANG)
    cmake_parse_arguments(RTIME "${options}" "${value_arg}" "${list_arg}" ${ARGN})

    IF (TARGET_ONLY)
        RETURN()
    ENDIF()

    FOREACH(lang ${RTIME_IFDOC_LANG})
        SET(doxlist)
        SET(includelist)
        FOREACH(ifdoc ${RTIME_IFDOC_SRC})
            STRING(REGEX REPLACE "\\.ifdoc" "" basename ${ifdoc})
            SET(output_dir
                "${RTI_MODULE_OUTPUT_SRC}/doxygen/${lang}/aliases/${RTIME_IFDOC_DSTDIR}"
            )
            SET(output_file "${output_dir}/${basename}.dox")
            ADD_CUSTOM_COMMAND(
                OUTPUT ${output_file}
                COMMAND ${CMAKE_COMMAND} -E make_directory ${output_dir}
                COMMAND ${DOC_JDK_HOME}/bin/java -cp ${XALANJAR} ${PROCESSOR}
            	    -in ${RTI_MODULE_SRCROOT}/srcIfdoc/${RTIME_IFDOC_SRCDIR}/${basename}.ifdoc
            	    -xsl ${RTIME_RESOURCE_ROOT}/ifdoc/srcXsl/ifdocToAliases${lang}.xsl
            	    -out ${output_file}
                DEPENDS ${RTI_MODULE_SRCROOT}/srcIfdoc/${RTIME_IFDOC_SRCDIR}/${basename}.ifdoc
            )
            SET_SOURCE_FILES_PROPERTIES(${output_file}
                                        PROPERTIES GENERATED 1
            )
            LIST(APPEND doxlist
                ${output_file}
            )
            SET(includelist
                 "${includelist}@INCLUDE = ${output_file}\n")
        ENDFOREACH()
        GET_PROPERTY(xl GLOBAL
                     PROPERTY ${RTI_MODULE_NAME}_DOX_${lang}_ALIASES
        )

        SET_PROPERTY(GLOBAL
                     PROPERTY ${RTI_MODULE_NAME}_DOX_${lang}_ALIASES ${doxlist} ${xl}
        )

        GET_PROPERTY(l GLOBAL
                     PROPERTY ${RTI_MODULE_NAME}_DOX_${lang}_LIST
        )

        SET_PROPERTY(GLOBAL
                     PROPERTY ${RTI_MODULE_NAME}_DOX_${lang}_LIST "${includelist} ${l}"
        )
    ENDFOREACH()

    SET(dir_count 0)

    IF (DEFINED RTIME_IFDOC_DIR)
        LIST(LENGTH RTIME_IFDOC_DIR dir_count)
    ENDIF()

    IF (${dir_count} EQUAL 0)
        FOREACH(lang ${RTIME_IFDOC_LANG})

            GET_PROPERTY(includelist GLOBAL
                         PROPERTY ${RTI_MODULE_NAME}_DOX_${lang}_LIST
            )

            # Note the indentation. Keep aaditional lines aligned to the 2nd line,
            # this results in nicer generated files.
            STRING(TOUPPER ${RTI_MODULE_NAME} RTI_MODULE_NAME_UC)
            SET(RTIME_DLL_EXPORT_DEFINE
                "\"${RTI_MODULE_NAME_UC}DllExport=\" \\ \n\
        \"${RTI_MODULE_NAME_UC}DllVariable=\" \n")

            SET(RTIME_INCLUDE_DEFINE ${includelist})
            CONFIGURE_FILE(
                ${RTIME_RESOURCE_ROOT}/doxygen/module_public.dox.in
                ${RTI_MODULE_OUTPUT_SRC}/doxygen/${lang}/${RTI_MODULE_NAME}_public.dox
            )
        ENDFOREACH()
    ENDIF()

ENDFUNCTION()

                                    IFDOC_DSTDIR ${dest_dir}

    FOREACH(lang ${RTIME_IFDOC_LANG})

        GET_PROPERTY(includelist GLOBAL
                     PROPERTY ${RTI_MODULE_NAME}_DOX_${lang}_LIST
        )

        # Note the indentation. Keep aaditional lines aligned to the 2nd line,
        # this results in nicer generated files.

        STRING(REPLACE "_" "" RTI_MODULE_NAME_UC_NO_UL ${RTI_MODULE_NAME_UC})
        SET(RTIME_DLL_EXPORT_DEFINE
            "\"${RTI_MODULE_NAME_UC}DllExport=\" \\ \n\
\"${RTI_MODULE_NAME_UC}DllVariable=\" \\ \n\

\"${RTI_MODULE_NAME_UC_NO_UL}DllVariable=\" \\ \n")

            ${RTIME_RESOURCE_ROOT}/doxygen/module_public.dox.in
            ${RTI_MODULE_OUTPUT_SRC}/doxygen/${lang}/${RTI_MODULE_NAME}_public.dox
        )
    ENDFOREACH()

#===============================================================================
# FUNCTION ....: rtime_add_doc
# PARAMETERS ..:
# SYNOPSIS ....: Generate target to build documentation
#
# doc - All documenatation
# doc_html - All HTML documentation
# doc_pdf - All PDF documentation
# doc_um - Only user manual
# doc_um_html - User manual html
# doc_um_pdf - User manual PDF
# doc_api - All API reference manuals
# doc_api_html - All API reference manuals in HTML
# doc_api_pdf -  All API reference manuals in PDF
# doc_security_sdk - Security SDK documentation
# dox_clean - Remove all doxygen alias files
#
# The following environment variable are applicable and use if set:
#
# DOC_JDK_HOME=<path to jdk1.8.0_121>
# XALANJAR=<path to xalan jar file>
# MAKEDOC=<path> to doxygen
# MAKE_EXEC=<make executable>
#
# Used indirectly by the generated Makefile for sphinx:
# SPHINXBUILD="path to sphinx-build"
#===============================================================================
FUNCTION(GET_PRODUCT_NAME inname outname)
    STRING(REPLACE "rti_me_([a-zA-Z_])\\.[0-9]+\\.[0-9]+" "\\1"
                        ${outname} ${inname})
    STRING(REGEX REPLACE "rti_me_([a-zA-Z_]+)\\.[0-9]+\\.[0-9]+" "\\1"
                        ${outname} ${inname})
    SET(${outname} ${${outname}} PARENT_SCOPE)
ENDFUNCTION(GET_PRODUCT_NAME)

FUNCTION(rtime_build_doc)
    SET(options)
    SET(value_arg PRODUCT PRODUCT_DOCNAME)
    SET(list_arg MODULES)
    cmake_parse_arguments(RTIME "${options}" "${value_arg}" "${list_arg}" ${ARGN})

    IF (NOT RTIME_PRODUCT)
        SET(RTIME_PRODUCT "rti_me.2.0")
        SET(RTIME_PRODUCT_DOCDIR "")
        SET(RTIME_PRODUCT_TARGET "_rti_me")
    ELSE()
        GET_PRODUCT_NAME(${RTIME_PRODUCT} product)
        SET(RTIME_PRODUCT_DOCDIR "/${product}")
        SET(RTIME_PRODUCT_TARGET "_${product}")
    ENDIF()

    IF (NOT RTIME_PRODUCT_DOCNAME)
       SET(RTIME_PRODUCT_DOCNAME "ConnextMicro")
    ENDIF()

    IF (NOT RTIME_MODULES)
        SET(DOC_MODULE_LIST ${MODULE_LIST})
    ELSE()
        SET(DOC_MODULE_LIST ${RTIME_MODULES})
    ENDIF()

    FOREACH(lang C Cpp)
        SET(doxygen_alias_list)
        SET(doxygen_root_list)
        FOREACH(min ${DOC_MODULE_LIST})
            GET_MODULE_NAME(${min} m)
            GET_PROPERTY(doxlist GLOBAL
                         PROPERTY ${m}_DOX_${lang}_ALIASES
                         SET)

            IF (${doxlist})
                GET_PROPERTY(doxlist GLOBAL
                             PROPERTY ${m}_DOX_${lang}_ALIASES
                )
                LIST(APPEND doxygen_alias_list ${doxlist})
            ENDIF()
        ENDFOREACH()
        ADD_CUSTOM_TARGET(doc${RTIME_PRODUCT_TARGET}_${lang}
                          DEPENDS ${doxygen_alias_list}
                          WORKING_DIRECTORY ${RTIME_SOURCE_ROOT}
        )
    ENDFOREACH()

    FOREACH(lang C Cpp)
        SET(module_doclist)
        SET(module_dirlist)
        STRING(TOLOWER ${lang} lang_lc)
        FOREACH (m ${DOC_MODULE_LIST})
            GET_MODULE_NAME(${m} mn)
            GET_PROPERTY(doxlist GLOBAL
                         PROPERTY ${mn}_DOX_${lang}_ALIASES
                         SET)
            IF(doxlist)
                SET(module_doclist
                "${module_doclist}@INCLUDE =
                ${RTIME_BUILD_ROOT}/${m}/doxygen/${lang}/${mn}_public.dox\n")
                LIST(APPEND module_dirlist "${RTIME_BUILD_ROOT}/${m}/doxygen")
            ENDIF()
        ENDFOREACH()
        CONFIGURE_FILE(
            ${RTIME_PRODUCT}/doc/API/apis/${lang}_public.dox.in
            ${RTIME_BUILD_ROOT}/rti_me.2.0/build/doc${RTIME_PRODUCT_DOCDIR}/API/apis/${lang}_public.dox
        )

        ADD_CUSTOM_TARGET(dox${RTIME_PRODUCT_TARGET}_${lang}_clean
              COMMAND ${CMAKE_COMMAND} -E remove_directory ${module_dirlist}
        )

        ADD_CUSTOM_TARGET(dox${RTIME_PRODUCT_TARGET}_${lang}_html
          COMMAND ${CMAKE_COMMAND} -E remove_directory
                    ${RTIME_BUILD_ROOT}/rti_me.2.0/doc${RTIME_PRODUCT_DOCDIR}/api_${lang_lc}
          COMMAND ${CMAKE_COMMAND} -E make_directory
                    ${RTIME_BUILD_ROOT}/rti_me.2.0/doc${RTIME_PRODUCT_DOCDIR}/api_${lang_lc}
          COMMAND ${RTIME_RESOURCE_ROOT}/scripts/rtime-doxygen
               --doxygen-path ${MAKEDOC}
               --rtime-root ${RTIME_SOURCE_ROOT}
               --rtime-resource ${RTIME_RESOURCE_ROOT}
               --output-dir ${RTIME_BUILD_ROOT}/rti_me.2.0/doc${RTIME_PRODUCT_DOCDIR}/api_${lang_lc}
               --main ${RTIME_BUILD_ROOT}/rti_me.2.0/build/doc${RTIME_PRODUCT_DOCDIR}/API/apis/${lang}_public.dox
          DEPENDS doc${RTIME_PRODUCT_TARGET}_${lang}
          WORKING_DIRECTORY ${RTIME_SOURCE_ROOT}
        )
        LIST(APPEND DOC_TARGETS_HTML dox${RTIME_PRODUCT_TARGET}_${lang}_html)
    ENDFOREACH()

    ADD_CUSTOM_TARGET(dox${RTIME_PRODUCT_TARGET}_clean
        DEPENDS dox${RTIME_PRODUCT_TARGET}_C_clean dox${RTIME_PRODUCT_TARGET}_Cpp_clean
    )
    LIST(APPEND DOC_TARGETS_CLEAN dox${RTIME_PRODUCT_TARGET}_clean)

    # Tweak the destination for the top-level Readme since it is not
    # symmetric
    IF ("${RTIME_PRODUCT}" STREQUAL "rti_me.2.0")
        SET(README_DST "")
    ELSE()
        SET(README_DST "doc/${RTIME_PRODUCT_DOCDIR}")
    ENDIF()

    ADD_CUSTOM_TARGET(doc${RTIME_PRODUCT_TARGET}_um_html
        COMMAND ${CMAKE_COMMAND} -E remove_directory
                ${RTIME_BUILD_ROOT}/rti_me.2.0/doc${RTIME_PRODUCT_DOCDIR}/doctrees
        COMMAND ${CMAKE_COMMAND} -E remove_directory
                ${RTIME_BUILD_ROOT}/rti_me.2.0/doc${RTIME_PRODUCT_DOCDIR}/html
        COMMAND make -C ${RTIME_SOURCE_ROOT}/${RTIME_PRODUCT}/doc/UM html
                BUILDDIR=${RTIME_BUILD_ROOT}/rti_me.2.0/doc${RTIME_PRODUCT_DOCDIR}/
        COMMAND ${CMAKE_COMMAND} -E remove_directory
                ${RTIME_BUILD_ROOT}/rti_me.2.0/doc${RTIME_PRODUCT_DOCDIR}/doctrees
        COMMAND ${CMAKE_COMMAND} -E copy
                ${RTIME_SOURCE_ROOT}/${RTIME_PRODUCT}/doc/ReadMe.html
                ${RTIME_BUILD_ROOT}/rti_me.2.0/${README_DST}
        WORKING_DIRECTORY ${RTIME_SOURCE_ROOT}
    )
    LIST(APPEND DOC_TARGETS_HTML doc${RTIME_PRODUCT_TARGET}_um_html)

    ADD_CUSTOM_TARGET(doc${RTIME_PRODUCT_TARGET}_um_pdf
        COMMAND ${MAKE_EXEC} -C ${RTIME_SOURCE_ROOT}/${RTIME_PRODUCT}/doc/UM pdf
                BUILDDIR=${RTIME_BUILD_ROOT}/rti_me.2.0/doc${RTIME_PRODUCT_DOCDIR}/pdf
        WORKING_DIRECTORY ${RTIME_SOURCE_ROOT}
    )
    LIST(APPEND DOC_TARGETS_PDF doc${RTIME_PRODUCT_TARGET}_um_pdf)

    ADD_CUSTOM_TARGET(doc${RTIME_PRODUCT_TARGET}_um
        DEPENDS doc${RTIME_PRODUCT_TARGET}_um_html doc${RTIME_PRODUCT_TARGET}_um_pdf
        WORKING_DIRECTORY ${RTIME_SOURCE_ROOT}
    )

    ADD_CUSTOM_TARGET(doc${RTIME_PRODUCT_TARGET}_api_html
        DEPENDS doc${RTIME_PRODUCT_TARGET}_C dox${RTIME_PRODUCT_TARGET}_C_html doc${RTIME_PRODUCT_TARGET}_Cpp dox${RTIME_PRODUCT_TARGET}_Cpp_html
        WORKING_DIRECTORY ${RTIME_SOURCE_ROOT}
    )
    LIST(APPEND DOC_TARGETS_HTML doc${RTIME_PRODUCT_TARGET}_api_html)

    FOREACH(lang C Cpp)
        STRING(TOLOWER ${lang} lang_lc)
        ADD_CUSTOM_TARGET(doc${RTIME_PRODUCT_TARGET}_api_${lang_lc}_pdf
            COMMAND ${CMAKE_COMMAND} -E make_directory
                    ${RTIME_BUILD_ROOT}/rti_me.2.0/doc${RTIME_PRODUCT_DOCDIR}/pdf
            COMMAND ${MAKE_EXEC} -C ${RTIME_BUILD_ROOT}/rti_me.2.0/doc${RTIME_PRODUCT_DOCDIR}/api_${lang_lc}/latex pdf
            COMMAND ${CMAKE_COMMAND} -E copy
                ${RTIME_BUILD_ROOT}/rti_me.2.0/doc${RTIME_PRODUCT_DOCDIR}/api_${lang_lc}/latex/refman.pdf
                ${RTIME_BUILD_ROOT}/rti_me.2.0/doc${RTIME_PRODUCT_DOCDIR}/pdf/RTI_${RTIME_PRODUCT_DOCNAME}_${lang}_API.pdf
            DEPENDS dox${RTIME_PRODUCT_TARGET}_${lang}_html
            WORKING_DIRECTORY ${RTIME_SOURCE_ROOT}
        )
        LIST(APPEND DOC_TARGETS_PDF doc${RTIME_PRODUCT_TARGET}_api_${lang_lc}_pdf)
    ENDFOREACH()

    ADD_CUSTOM_TARGET(doc${RTIME_PRODUCT_TARGET}_api_pdf
        DEPENDS doc${RTIME_PRODUCT_TARGET}_api_c_pdf doc${RTIME_PRODUCT_TARGET}_api_cpp_pdf
        WORKING_DIRECTORY ${RTIME_SOURCE_ROOT}
    )
    LIST(APPEND DOC_TARGETS_PDF doc${RTIME_PRODUCT_TARGET}_api_pdf)

    ADD_CUSTOM_TARGET(doc${RTIME_PRODUCT_TARGET}_html
        DEPENDS doc${RTIME_PRODUCT_TARGET}_um_html doc${RTIME_PRODUCT_TARGET}_api_html
        WORKING_DIRECTORY ${RTIME_SOURCE_ROOT}
    )
    LIST(APPEND DOC_TARGETS_HTML doc${RTIME_PRODUCT_TARGET}_html)

    ADD_CUSTOM_TARGET(doc${RTIME_PRODUCT_TARGET}_pdf
        DEPENDS doc${RTIME_PRODUCT_TARGET}_um_pdf doc${RTIME_PRODUCT_TARGET}_api_pdf
        WORKING_DIRECTORY ${RTIME_SOURCE_ROOT}
    )
    LIST(APPEND DOC_TARGETS_PDF doc${RTIME_PRODUCT_TARGET}_pdf)

    ADD_CUSTOM_TARGET(doc${RTIME_PRODUCT_TARGET}
        DEPENDS doc${RTIME_PRODUCT_TARGET}_html doc${RTIME_PRODUCT_TARGET}_pdf
        WORKING_DIRECTORY ${RTIME_SOURCE_ROOT}
    )
    LIST(APPEND DOC_TARGETS_PRODUCT doc${RTIME_PRODUCT_TARGET})

    SET(DOC_TARGETS_HTML ${DOC_TARGETS_HTML} PARENT_SCOPE)
    SET(DOC_TARGETS_PRODUCT ${DOC_TARGETS_PRODUCT} PARENT_SCOPE)
    SET(DOC_TARGETS_PDF ${DOC_TARGETS_PDF} PARENT_SCOPE)

ENDFUNCTION()

function(copy_example)
    SET(options)
    SET(value_arg EXAMPLE_SRCDIR EXAMPLE_DSTDIR EXAMPLE_NEWLINE)
    SET(list_arg EXAMPLE_FILELIST EXAMPLE_DIRLIST)
    cmake_parse_arguments(RTIME "${options}" "${value_arg}" "${list_arg}" ${ARGN})

    FOREACH(f ${RTIME_EXAMPLE_FILELIST})
        configure_file(${RTIME_EXAMPLE_SRCDIR}/${f}
                       ${RTIME_EXAMPLE_DSTDIR}/${f}
                       @ONLY NEWLINE_STYLE ${RTIME_EXAMPLE_NEWLINE})
    ENDFOREACH()
    FOREACH(d ${RTIME_EXAMPLE_DIRLIST})
        string(REGEX REPLACE "/" ";" dirlist ${d})
        list(LENGTH dirlist length)
        MATH(EXPR length "${length} - 1")
        list(GET dirlist ${length} leafdir)
        EXECUTE_PROCESS(COMMAND ${CMAKE_COMMAND}
                    -E copy_directory ${RTIME_SOURCE_ROOT}/${d}
                                      ${RTIME_EXAMPLE_DSTDIR}/${leafdir})
    ENDFOREACH()
endfunction()

function(add_product_example)
    SET(options)
    SET(value_arg PRODUCT)
    SET(list_arg EXAMPLES)

    cmake_parse_arguments(RTIME "${options}" "${value_arg}" "${list_arg}" ${ARGN})

    # These targets don't do anything for the time being, but is available
    # so the following can be executed:
    # ./resource.1.0/scripts/rtime-make --target self --name examples --build -- examples
    SET(PRODUCT_TARGET examples_${RTIME_PRODUCT})

    add_custom_target(${PRODUCT_TARGET})

    SET(example_root "${RTIME_SOURCE_ROOT}/${RTIME_PRODUCT}/example")
    foreach(ex ${RTIME_EXAMPLES})
        foreach(lf unix:UNIX windows:WIN32)
            IF (EXISTS ${example_root}/${ex}/CMakeExampleLists.txt)
                INCLUDE(${example_root}/${ex}/CMakeExampleLists.txt)
                IF (${lf} MATCHES "(.*):(.*)")
                    copy_example(
                        EXAMPLE_SRCDIR
                            ${example_root}/${ex}
                         EXAMPLE_DSTDIR
                            ${RTIME_MODULE_EXPORT_ROOT}/example/${CMAKE_MATCH_1}/${ex}
                         EXAMPLE_FILELIST
                            ${FILE_LIST_COMMON}
                         EXAMPLE_DIRLIST
                            ${COPY_DIRS}
                         EXAMPLE_NEWLINE
                            ${CMAKE_MATCH_2}
                    )
                ELSE()
                    RTI_MESSAGE_FATAL("no newline style found for the examples")
                ENDIF()
            ENDIF()
        endforeach()
    endforeach()
    SET(EXAMPLE_TARGETS "${EXAMPLE_TARGETS};${PRODUCT_TARGET}" PARENT_SCOPE)
endfunction()

function(add_exported_header_dependency)
    set(options)
    set(value_arg DEPEDENCY)
    set(list_arg DEPENDENTS)
    cmake_parse_arguments(RTIME "${options}" "${value_arg}" "${list_arg}" ${ARGN})

    set(try_names ${RTIME_DEPEDENCY} lib${RTIME_DEPEDENCY})
    string(REGEX MATCH "_" m ${RTIME_DEPEDENCY})
    if (m)
        string(REPLACE "_" "" s ${RTIME_DEPEDENCY})
        list(APPEND try_names ${s} lib${s})
    endif()

    foreach(n ${try_names})
        get_property(exp_hdr GLOBAL PROPERTY ${n}_EXPORTED_HEADER_TARGET)
        if (DEFINED exp_hdr)
            break()
        endif()
    endforeach()

    if (DEFINED exp_hdr)
        foreach(d ${RTIME_DEPENDENTS})
            add_dependencies(${d} ${exp_hdr})
        endforeach()
    endif()
endfunction()

function(add_header_export)
    set(options)
    set(value_arg MODULE_NAME TARGET_LIB INPUT_FILE OUTPUT_FILE )
    set(list_arg)
    cmake_parse_arguments(F "${options}" "${value_arg}" "${list_arg}" ${ARGN})

    GET_PROPERTY(p GLOBAL PROPERTY ${F_OUTPUT_FILE} SET)
    if (p)
        return()
    endif()

    add_custom_command(
        TARGET ${F_TARGET_LIB} POST_BUILD
        COMMAND
            ${CMAKE_COMMAND} -E copy_if_different ${F_INPUT_FILE} ${F_OUTPUT_FILE}
    )

    add_custom_command(
        OUTPUT ${F_OUTPUT_FILE}
        COMMAND
            ${CMAKE_COMMAND} -E copy_if_different ${F_INPUT_FILE} ${F_OUTPUT_FILE}
        DEPENDS
            ${F_INPUT_FILE}
    )

    SET_PROPERTY(SOURCE ${F_OUTPUT_FILE} PROPERTY GENERATED true)
    SET_PROPERTY(SOURCE ${F_OUTPUT_FILE} PROPERTY RTI_GENERATED true)
    SET_PROPERTY(GLOBAL PROPERTY ${F_OUTPUT_FILE} TRUE)
endfunction()
