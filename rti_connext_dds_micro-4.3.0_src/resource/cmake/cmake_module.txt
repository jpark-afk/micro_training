#
# Build a module
#
# A module is a self-contained unit of code. Its public interface is
# defined in the <module>/include/<module> directory. Other modules shall never
# include header files from a submodule.
#
# A normal module:
# - Builds a shared and static library consisting of the source for
#   from each submodule
# - Build a module tester with test source from each submodule
# - Does not build each individual submodule as its own library
# - Makes all the object-files from the module available to be linked to a
#   different module (a meta-module)
#
# A module is the smallest unit of compilation. A module produces one or more
# of the following:
# - A module library - A module library contains source from all the submodules
# - A library        - A library is built from source in the same directory
# - An executable    - An executable is built from source in the directory
#
#
# The following global variables can be set:
#
# RTI_BUILD_UNITTESTS - Only build the unit-tests if true
#
# The following variables can be set at the module level:
#
# RTI_MODULE_DEPENDS  - A list of modules that this module depends on. For each
#                      module listed:
#                        - The module's DLL_VARIABLE is defined
#                        - The module's public header directory is added to the
#                          include path (_build_ directory)
#
# RTI_LIBRARY_DEPENDS  - A list of libraries this module will link against.
#                       A module may create multiple libraries, or libraries
#                       with a different name. Thus, it is necessary to list
#                       all of them. These are linked from the _build_
#                       directory
#
# RTI_OBJECT_DEPENDS   - A list of object libraries to add to the module
#
# RTI_MODULE_IMPORTS   - RTI_MODULE_DEPEND + RTI_LIBRARY_DEPEND
#
# RTI_MODULE_TEST_IMPORT   - Does the same as RTI_MODULE_IMPORT, but also:
#                           - Automatically adds that modules library as a
#                             dependency. The same can be achieved with
#                             a RTI_MODULE_IMPORT and RTI_LIBRARY_IMPORT
#
# PLATFORM_TEST_LIBS  - Libraries added to unit-tests in addition to
#                       PLATFORM_LIBS
#
# RTI_MODULE_EXPORT   - Export this module with the given library name
#
# RTI_SUBMODULES      - List of submodules as part of this module. Submodules
#                       can only depend on a submodule before it, not after.
#                       A submodule only contributes code to the module library,
#                       it does not produce any artifacts on its own.
#
################################################################################

                    LIST(APPEND ${RTI_MODULE_NAME}_PUBLIC_HDR

                    LIST(APPEND ${RTI_MODULE_NAME}_PUBLIC_HDR
                                ${RTI_MODULE_SRCROOT}/include/${sd}/${f})

                        LIST(APPEND ${RTI_MODULE_NAME}_EXPORT ${RTI_SHARED_MODULE_EXPORT_DIR}${sd}/${f})
                    ELSE()
                        LIST(APPEND ${RTI_MODULE_NAME}_EXPORT ${RTI_MODULE_NAME}/${f})
                    ENDIF()
                ELSE()
                    LIST(APPEND ${RTI_MODULE_NAME}_PUBLIC_HDR
                                ${CMAKE_CURRENT_SOURCE_DIR}/include/${RTI_MODULE_NAME}/${shared_module}/${f})
                    LIST(APPEND ${RTI_MODULE_NAME}_EXPORT ${RTI_MODULE_NAME}/${f})

            LIST(APPEND ${RTI_MODULE_NAME}_EXPORT ${RTI_MODULE_NAME}/${f})
        ENDFOREACH()
        IF (RTI_MODULE_PUBLIC_HDR_FOLDER)
            LIST(APPEND RTI_MODULE_INCLUDE_DIRS
                 ${RTI_MODULE_SRCROOT}/include/${RTI_MODULE_NAME})
        ENDIF()
        FOREACH(f ${RTI_MODULE_PUBLIC_HDR_FOLDER})
            IF (EXISTS ${RTI_MODULE_SRCROOT}/include/${RTI_MODULE_NAME}/${f}/CMakeLists.txt)
                SET(RTI_MODULE_PUBLIC_HDR_TMP ${RTI_MODULE_PUBLIC_HDR})
                INCLUDE(${RTI_MODULE_SRCROOT}/include/${RTI_MODULE_NAME}/${f}/CMakeLists.txt)
                FOREACH(h ${RTI_MODULE_PUBLIC_HDR})
                    LIST(APPEND RTI_MODULE_PUBLIC_HDR_TMP ${f}/${h})
                    LIST(APPEND ${RTI_MODULE_NAME}_PUBLIC_HDR
                                ${RTI_MODULE_SRCROOT}/include/${RTI_MODULE_NAME}/${f}/${h})
                    LIST(APPEND ${RTI_MODULE_NAME}_EXPORT ${RTI_MODULE_NAME}/${f}/${h})
                ENDFOREACH()
                SET(RTI_MODULE_PUBLIC_HDR ${RTI_MODULE_PUBLIC_HDR_TMP})
            ENDIF()
        ENDFOREACH()
    ENDIF()

    # Add fully qualified path sources for public headers
    FOREACH(f ${RTI_MODULE_PUBLIC_HDR})
        LIST(APPEND ${RTI_MODULE_NAME}_PUBLIC_HDR
                    ${RTI_MODULE_SRCROOT}/include/${RTI_MODULE_NAME}/${f})
        LIST(APPEND ${RTI_MODULE_NAME}_EXPORT ${RTI_MODULE_NAME}/${f})
    ENDFOREACH()
    SET(${RTI_MODULE_NAME}_EXPORT_HDR ${${RTI_MODULE_NAME}_EXPORT} PARENT_SCOPE)

    SET(RTIME_SHIP_SOURCE TRUE)
    IF (EXISTS ${RTIME_SOURCE_ROOT}/include/rti_me/${RTI_MODULE_NAME}/CMakeLists.txt
        OR IS_SYMLINK ${RTIME_SOURCE_ROOT}/include/rti_me/${RTI_MODULE_NAME}/CMakeLists.txt)

            LIST(APPEND ${RTI_MODULE_NAME}_PUBLIC_HDR
                        ${RTIME_SOURCE_ROOT}/include/rti_me/${RTI_MODULE_NAME}/${f})
            LIST(APPEND ${RTI_MODULE_NAME}_EXPORT ${RTI_MODULE_NAME}/${f})
        ENDIF()
    ENDFOREACH()

    IF (RTIME_FULL_SOURCE AND NOT RTIME_DEVTREE)
        FOREACH(f ${RTI_MODULE_PUBLIC_HDR} ${RTI_MODULE_SHARED_HDR}
        ${RTI_MODULE_SHARED_HDR_PATCHED})

                           "${RTIME_SOURCE_ROOT}/include/rti_me/${f}"
                        DESTINATION
                            "include/rti_me"
                )
            ELSEIF(${f} MATCHES "rti_me_psl_dll.h")
                INSTALL(FILES
                           "${RTIME_SOURCE_ROOT}/include/rti_me/${RTI_MODULE_NAME}/${f}"
                        DESTINATION
                            "include/rti_me/${RTI_MODULE_NAME}"
                )
            ELSEIF(${f} MATCHES "rti_me")
                INSTALL(FILES
                            "${RTIME_SOURCE_ROOT}/include/rti_me/${f}"
                        DESTINATION
                            "include/rti_me"
                )
            ELSE()
                INSTALL(FILES
                            "${RTIME_SOURCE_ROOT}/include/rti_me/${RTI_MODULE_NAME}/${f}"
                        DESTINATION
                            "include/rti_me/${RTI_MODULE_NAME}"
                )
            ENDIF()
        ENDFOREACH()
    ENDIF()

    FOREACH(f ${RTI_MODULE_PUBLIC_HDR_FOLDER})
        IF (EXISTS ${RTIME_SOURCE_ROOT}/include/rti_me/${RTI_MODULE_NAME}/${f}/CMakeLists.txt
        OR IS_SYMLINK ${RTIME_SOURCE_ROOT}/include/rti_me/${RTI_MODULE_NAME}/${f}/CMakeLists.txt)
            SET(RTI_MODULE_PUBLIC_HDR_TMP ${RTI_MODULE_PUBLIC_HDR})
            INCLUDE(${RTIME_SOURCE_ROOT}/include/rti_me/${RTI_MODULE_NAME}/${f}/CMakeLists.txt)
            FOREACH(h ${RTI_MODULE_PUBLIC_HDR})
                LIST(APPEND RTI_MODULE_PUBLIC_HDR_TMP ${f}/${h})
                LIST(APPEND ${RTI_MODULE_NAME}_PUBLIC_HDR

                LIST(APPEND ${RTI_MODULE_NAME}_EXPORT
                ${RTI_MODULE_NAME}/${f}/${h})
                IF (RTIME_FULL_SOURCE AND NOT RTIME_DEVTREE)
                    INSTALL(FILES
                                "${RTIME_SOURCE_ROOT}/include/rti_me/${RTI_MODULE_NAME}/${f}/${h}"
                            DESTINATION
                                "include/rti_me/${RTI_MODULE_NAME}/${f}"
                    )
                ENDIF()
            ENDFOREACH()
            SET(RTI_MODULE_PUBLIC_HDR ${RTI_MODULE_PUBLIC_HDR_TMP})
        ENDIF()
    ENDFOREACH()
    SET(RTIME_SHIP_SOURCE FALSE)

                    INSTALL(FILES
                        "${RTIME_SOURCE_ROOT}/include/rti_me/${d}/${f}"
                    DESTINATION
                        "include/rti_me/${d}"
                    )
                ENDFOREACH()
            ENDIF()
        ENDIF()
    ENDFOREACH()

# Add each submodule. Note that it is not required that each submodule
# contributes to the "module" library. This is because a submoule may be
# its own execuable or its own library. This is determined by the included
# CMakeLists.txt file
################################################################################

# For each submodule, potential source and headers to the module library
SET(SUBMODULE_GENERATE)
SET(SUBMODULE_HAS_GENERATE false)
SET(SUBMODULE_TEST_DEPENDS)

FOREACH(srcdir "" /srcC /srcCxx)
    FOREACH(m ${RTI_SUBMODULES})
        UNSET(RTI_SUBMODULE_TEST_QUACK_TS)
        UNSET(RTI_SUBMODULE_TEST_QUACK_TI)
        UNSET(RTI_SUBMODULE_TEST_QUACK_TS_GEN)
        UNSET(RTI_SUBMODULE_TEST_SRC)
        UNSET(RTI_SUBMODULE_TEST_SRC_CC)
        UNSET(RTI_SUBMODULE_TEST_HDR)
        UNSET(IDL_SOURCE)
        UNSET(IDL_SOURCE_CXX)
        UNSET(RTI_SUBMODULE_SRC)
        UNSET(RTI_SUBMODULE_SRC_CC)
        UNSET(RTI_SUBMODULE_HDR)
        UNSET(RTI_SUBMODULE_IDL)
        UNSET(RTI_SUBMODULE_TEST_IDL)
        UNSET(RTI_SHARED_SRC)
        UNSET(RTI_SHARED_HDR)
        UNSET(RTI_SUBMODULE_XML_APPGEN)
        UNSET(RTI_SHARED_SUBMODULE)
        SET(RTI_SUBMODULE_NAME ${m})
        UNSET(RTI_SUBMODULE_QUACK_TS)
        UNSET(RTI_SUBMODULE_QUACK_TI)
        UNSET(RTI_SUBMODULE_QUACK_TS_GEN)

        # Add C source
        IF (EXISTS ${RTI_MODULE_SRCROOT}${srcdir}/${m}/CMakeLists.txt)
            IF (RTI_ADD_SUBMOUDLE_INCLUDE)
                INCLUDE_DIRECTORIES(${RTI_MODULE_SRCROOT}${srcdir}/${m})
            ELSE()
                LIST(APPEND RTI_MODULE_INCLUDE_DIRS ${RTI_MODULE_SRCROOT}${srcdir}/${m})
            ENDIF()
            INCLUDE (${RTI_MODULE_SRCROOT}${srcdir}/${m}/CMakeLists.txt)
        ENDIF()

        # Add the submodule test directory if it exists
        IF (EXISTS ${RTI_MODULE_SRCROOT}${srcdir}/${m}/test/CMakeLists.txt)
            LIST(APPEND RTI_MODULE_INCLUDE_TEST_DIRS
                        ${RTI_MODULE_SRCROOT}${srcdir}/${m}/test)
            INCLUDE (${RTI_MODULE_SRCROOT}${srcdir}/${m}/test/CMakeLists.txt)
        ENDIF()

        GET_PROPERTY(sm_generate GLOBAL
                     PROPERTY ${RTI_MODULE_NAME}_${RTI_SUBMODULE_NAME}_HAS_GENERATE
                     SET)

        IF (sm_generate AND ${sm_generate})
            SET(SUBMODULE_HAS_GENERATE true)
            SET(SUBMODULE_GENERATE ${SUBMODULE_GENERATE} ${RTI_MODULE_NAME}_${RTI_SUBMODULE_NAME}_GENERATE)
        ENDIF()

    ENDFOREACH()

    # Add the main test directory if it exists
    IF (EXISTS ${RTI_MODULE_SRCROOT}${srcdir}/test/CMakeLists.txt)
        LIST(APPEND RTI_MODULE_INCLUDE_TEST_DIRS
                    ${RTI_MODULE_SRCROOT}${srcdir}/test)
    ENDIF()

ENDFOREACH()

IF (DEFINED RTI_MODULE_INCLUDE_DIRS)
    LIST(REMOVE_DUPLICATES RTI_MODULE_INCLUDE_DIRS)
ENDIF()

IF (DEFINED ${RTI_MODULE_NAME}_SRC_H)
    LIST(REMOVE_DUPLICATES ${RTI_MODULE_NAME}_SRC_H)
ENDIF()

IF (DEFINED ${RTI_MODULE_NAME}_PUBLIC_HDR)
    LIST(REMOVE_DUPLICATES ${RTI_MODULE_NAME}_PUBLIC_HDR)
ENDIF()

IF (DEFINED ${RTI_MODULE_NAME}_EXPORT)
    LIST(REMOVE_DUPLICATES ${RTI_MODULE_NAME}_EXPORT)
ENDIF()

SET(${RTI_MODULE_NAME}_INTERFACE ${${RTI_MODULE_NAME}_PUBLIC_HDR} PARENT_SCOPE)
SET(${RTI_MODULE_NAME}_SOURCE    ${${RTI_MODULE_NAME}_SRC_C} PARENT_SCOPE)
SET(${RTI_MODULE_NAME}_HEADER    ${${RTI_MODULE_NAME}_SRC_H} PARENT_SCOPE)

LIST(LENGTH ${RTI_MODULE_NAME}_SRC_C el)
IF (${el} GREATER 0)
    SET(RTI_BUILD_MODULE_HAS_SOURCE TRUE)
ELSE()
    SET(RTI_BUILD_MODULE_HAS_SOURCE FALSE)
ENDIF()

# Add the depend libraries
SET(RTI_MODULE_DEPEND_SHARED_LIBS)
SET(RTI_MODULE_DEPEND_STATIC_LIBS)

FOREACH(md ${RTI_MODULE_IMPORTS} ${RTI_LIBRARY_DEPENDS})
    SET(IMPORT_STATIC_PREFIX ${CMAKE_STATIC_LIBRARY_PREFIX})
    SET(IMPORT_DYNAMIC_PREFIX ${CMAKE_SHARED_LIBRARY_PREFIX})

    IF (${md} MATCHES "(.*)::(.*)")
        set(md "${CMAKE_MATCH_2}")
    ENDIF()

    IF ((${md} MATCHES "lib.*") OR (${${md}_EXPORT_NAME} MATCHES "lib.*"))
        SET(IMPORT_STATIC_PREFIX "")
        SET(IMPORT_DYNAMIC_PREFIX "")
    ENDIF()

    IF (DEFINED ${md}_EXPORT_NAME)
        SET(RTI_MODULE_DEPEND_SHARED_LIB ${${md}_EXPORT_NAME}${RTI_LIB_SUFFIX_DYNAMIC})
        SET(RTI_MODULE_DEPEND_STATIC_LIB ${${md}_EXPORT_NAME}${RTI_LIB_SUFFIX_STATIC})

    ELSE()
        IF(${md} MATCHES "rti_me.*")
            SET(RTI_MODULE_DEPEND_SHARED_LIB ${md}${RTI_LIB_SUFFIX_DYNAMIC})
            SET(RTI_MODULE_DEPEND_STATIC_LIB ${md}${RTI_LIB_SUFFIX_STATIC})
        ELSE()
            SET(RTI_MODULE_DEPEND_SHARED_LIB ${RTI_LIB_PREFIX}_${md}${RTI_LIB_SUFFIX_DYNAMIC})
            SET(RTI_MODULE_DEPEND_STATIC_LIB ${RTI_LIB_PREFIX}_${md}${RTI_LIB_SUFFIX_STATIC})
        ENDIF()
    ENDIF()
    LIST(FIND RTI_PIL_DEPENDS ${md} found)
    IF (NOT ${found} EQUAL -1)
        if (NOT ${RTI_MODULE_DEPEND_STATIC_LIB}_IMPORTED)
            add_library(${RTI_MODULE_DEPEND_STATIC_LIB} STATIC IMPORTED)
            IF (NOT RTIME_PIL_PATH)
                SET(RTIME_PIL_PATH "${RTIME_MODULE_EXPORT_ROOT}/lib/${RTIME_PIL_ARCH}")
            ENDIF()
            set_property(TARGET ${RTI_MODULE_DEPEND_STATIC_LIB} PROPERTY
                         IMPORTED_LOCATION
                        ${RTIME_PIL_PATH}/${IMPORT_STATIC_PREFIX}${RTI_MODULE_DEPEND_STATIC_LIB}${CMAKE_STATIC_LIBRARY_SUFFIX}
            )
            add_library(${RTI_MODULE_DEPEND_SHARED_LIB} SHARED IMPORTED)

            set(location_property IMPORTED_LOCATION)
            set(extension ${CMAKE_SHARED_LIBRARY_SUFFIX})
            IF(WIN32)
                set(location_property IMPORTED_IMPLIB)
                set(extension ".lib")
            ENDIF()
            set_property(TARGET ${RTI_MODULE_DEPEND_SHARED_LIB} PROPERTY
                        ${location_property}
                        ${RTIME_PIL_PATH}/${IMPORT_DYNAMIC_PREFIX}${RTI_MODULE_DEPEND_SHARED_LIB}${extension}
            )
            SET(${RTI_MODULE_DEPEND_STATIC_LIB}_IMPORTED TRUE)
            SET(${RTI_MODULE_DEPEND_STATIC_LIB}_IMPORTED TRUE PARENT_SCOPE)
        ENDIF()
     ENDIF()
    LIST(APPEND RTI_MODULE_DEPEND_SHARED_LIBS ${RTI_MODULE_DEPEND_SHARED_LIB})
    LIST(APPEND RTI_MODULE_DEPEND_STATIC_LIBS ${RTI_MODULE_DEPEND_STATIC_LIB})
ENDFOREACH()

IF (NOT DEFINED RTI_OBJECT_MODULE)
    SET(RTI_OBJECT_MODULE FALSE)
ENDIF()

IF (${RTI_BUILD_MODULE_HAS_SOURCE} AND ${RTI_OBJECT_MODULE})
    SOURCE_GROUP(\\include\\${RTI_MODULE_NAME}
                 FILES ${${RTI_MODULE_NAME}_PUBLIC_HDR})
ENDIF()

################################################################################
#
# Create module libraries using the object libraries
#
################################################################################

IF (${RTI_BUILD_MODULE_HAS_SOURCE} AND
    ((NOT ${RTI_OBJECT_MODULE}) OR (${RTI_BUILD_UNITTESTS} AND (NOT RTIME_BUILD_SINGLE_LIB))))

    SET(RTI_SHARED_OBJLIB ${${RTI_MODULE_NAME}_PUBLIC_HDR}
        ${${RTI_MODULE_NAME}_SRC_C} ${${RTI_MODULE_NAME}_SRC_H})

    SET(RTI_STATIC_OBJLIB ${${RTI_MODULE_NAME}_PUBLIC_HDR}
        ${${RTI_MODULE_NAME}_SRC_C} ${${RTI_MODULE_NAME}_SRC_H})

    IF (RTI_BUILDID_MASTER AND (NOT RTIME_DISABLE_BUILDID))
        SET(BUILD_ID_FILE "${RTI_MODULE_OUTPUT_INCLUDE}/${RTI_MODULE_NAME}_buildid.h")

        IF (DEFINED RTIME_BUILD_ID)
            SET(RTI_BUILDID_DEFINE "-DRTIME_BUILD_ID=${RTIME_BUILD_ID}")
        ELSE()
            SET(RTI_BUILDID_DEFINE "")
        ENDIF()

        #LIST(APPEND RTI_SHARED_OBJLIB ${BUILD_ID_FILE})
        LIST(APPEND RTI_STATIC_OBJLIB ${BUILD_ID_FILE})

        # When there are multiple output files, CMake generated a dependency
        # between them based on the order in which they appear. So the below
        # creates a rule where BUILD_ID_FILE depends on PHONY_BUILDID.
        # However, PHONY_BUILDID generates BUILD_ID_FILE so it is always
        # out of date.
        ADD_CUSTOM_COMMAND(OUTPUT  ${RTI_MODULE_OUTPUT_INCLUDE}
                           COMMAND ${CMAKE_COMMAND} -E make_directory ${RTI_MODULE_OUTPUT_INCLUDE}
                           COMMENT "Generate ${RTI_MODULE_OUTPUT_INCLUDE}"
        )

        IF (RTI_MANUAL_BUILDID)
            ADD_CUSTOM_COMMAND(OUTPUT  ${BUILD_ID_FILE} ${RTI_MODULE_OUTPUT_INCLUDE}
                               COMMAND ${CMAKE_COMMAND} -DBUILD_ID_FILE=${BUILD_ID_FILE}
                                        -DHAS_STRING_TIMESTAMP=${HAS_STRING_TIMESTAMP}
                                        ${RTI_BUILDID_DEFINE}
                                        -P ${RTIME_RESOURCE_ROOT}/cmake/cmake_buildid.txt
                               COMMENT "Generate ${BUILD_ID_FILE}"
            )
            ADD_CUSTOM_TARGET(BUILDID
                              DEPENDS ${BUILD_ID_FILE}
            )
        ELSE()
            ADD_CUSTOM_COMMAND(OUTPUT  PHONY_BUILDID ${BUILD_ID_FILE} ${RTI_MODULE_OUTPUT_INCLUDE}
                               COMMAND ${CMAKE_COMMAND} -DBUILD_ID_FILE=${BUILD_ID_FILE}
                                        -DHAS_STRING_TIMESTAMP=${HAS_STRING_TIMESTAMP}
                                        ${RTI_BUILDID_DEFINE}
                                        -P ${RTIME_RESOURCE_ROOT}/cmake/cmake_buildid.txt
                               COMMENT "Generate ${BUILD_ID_FILE}"
            )
            ADD_CUSTOM_TARGET(BUILDID
                              DEPENDS PHONY_BUILDID
            )
        ENDIF()

        SET_SOURCE_FILES_PROPERTIES( PHONY_BUILDID PROPERTIES SYMBOLIC 1)
    ELSEIF((DEFINED RTI_BUILDID_MASTER) AND RTIME_DISABLE_BUILDID)
        ADD_CUSTOM_TARGET(BUILDID  )
    ENDIF()

    SET_SOURCE_FILES_PROPERTIES(${${RTI_MODULE_NAME}_PUBLIC_HDR}
                                 ${${RTI_MODULE_NAME}_SRC_C}
                                 ${${RTI_MODULE_NAME}_SRC_H}
                                 PROPERTIES
                                 COMPILE_DEFINITIONS RTI_MODULE_NAME=\"${RTI_MODULE_NAME}\")

    FOREACH (m ${RTI_OBJECT_DEPENDS})
        IF (${m} MATCHES "(.*)::(.*)")
            set(m "${CMAKE_MATCH_1}_${CMAKE_MATCH_2}")
        ENDIF()
        LIST(APPEND RTI_STATIC_OBJLIB
             ${${m}_INTERFACE} ${${m}_SOURCE} ${${m}_HEADER})
        LIST(APPEND RTI_SHARED_OBJLIB
             ${${m}_INTERFACE} ${${m}_SOURCE} ${${m}_HEADER})
        SET_SOURCE_FILES_PROPERTIES(${${m}_INTERFACE} ${${m}_SOURCE} ${${m}_HEADER}
             PROPERTIES
             COMPILE_DEFINITIONS RTI_MODULE_NAME=\"${m}\")
    ENDFOREACH()

    SET(${RTI_MODULE_NAME}_OBJECT_DEPENDS ${RTI_OBJECT_DEPENDS} PARENT_SCOPE)

    ADD_LIBRARY(${RTI_MODULE_LIBNAME_STATIC} STATIC  ${RTI_STATIC_OBJLIB})

    IF (NOT RTI_NO_SHARED_LIB)
        ADD_LIBRARY(${RTI_MODULE_LIBNAME_SHARED} SHARED ${RTI_SHARED_OBJLIB})
        IF (DEFINED RTI_BUILDID_MASTER)
            ADD_DEPENDENCIES(${RTI_MODULE_LIBNAME_SHARED}
                             ${RTI_MODULE_LIBNAME_STATIC})
        ENDIF()

        SET_TARGET_PROPERTIES(${RTI_MODULE_LIBNAME_SHARED}
                              PROPERTIES FOLDER ${CURRENT_MODULE_DIR})
        SET_TARGET_PROPERTIES(${RTI_MODULE_LIBNAME_SHARED}
                              PROPERTIES CMAKE_SKIP_RPATH TRUE)
    ENDIF()

    IF (RTI_MODULE_EXPORT AND RTIME_FULL_SOURCE AND NOT RTIME_DEVTREE)
        IF (NOT RTIME_INSTALL_DIR)
            SET(RTI_INSTALL_DIR "lib")
        ENDIF()
        INSTALL(TARGETS
                    ${RTI_MODULE_LIBNAME_STATIC}
                ARCHIVE
                    DESTINATION "${RTI_INSTALL_DIR}"
        )
        IF (NOT RTI_NO_SHARED_LIB)
            INSTALL(TARGETS
                       ${RTI_MODULE_LIBNAME_SHARED}
                    RUNTIME
                        DESTINATION "${RTI_INSTALL_DIR}"
                    LIBRARY
                        DESTINATION "${RTI_INSTALL_DIR}"
                    ARCHIVE
                        DESTINATION "${RTI_INSTALL_DIR}"
            )
        ENDIF()
    ENDIF()

    SET_TARGET_PROPERTIES(${RTI_MODULE_LIBNAME_STATIC}
                          PROPERTIES FOLDER ${CURRENT_MODULE_DIR})

    SOURCE_GROUP(\\include\\${RTI_MODULE_NAME}
                 FILES ${${RTI_MODULE_NAME}_PUBLIC_HDR})

    IF (DEFINED RTI_COMPILE_DEFINITIONS)
        IF (NOT RTI_NO_SHARED_LIB)
            TARGET_COMPILE_DEFINITIONS(${RTI_MODULE_LIBNAME_SHARED}
                                       PRIVATE ${RTI_COMPILE_DEFINITIONS})
        ENDIF()
        TARGET_COMPILE_DEFINITIONS(${RTI_MODULE_LIBNAME_STATIC}
                                   PRIVATE ${RTI_COMPILE_DEFINITIONS})
    ENDIF()

    IF (${HAS_TARGET_INCLUDE_DIR})
        IF (NOT RTI_NO_SHARED_LIB)
            TARGET_INCLUDE_DIRECTORIES(${RTI_MODULE_LIBNAME_SHARED}
                                       PRIVATE ${RTI_MODULE_INCLUDE_DIRS})
        ENDIF()
        TARGET_INCLUDE_DIRECTORIES(${RTI_MODULE_LIBNAME_STATIC}
                                   PRIVATE ${RTI_MODULE_INCLUDE_DIRS})
    ELSE()
        INCLUDE_DIRECTORIES(${RTI_MODULE_INCLUDE_DIRS})
    ENDIF()

    LIST(LENGTH RTI_MODULE_DEPEND_STATIC_LIBS x)
    IF (${x} GREATER 0)
        ADD_DEPENDENCIES(${RTI_MODULE_LIBNAME_STATIC}
                         ${RTI_MODULE_DEPEND_STATIC_LIBS})
    ENDIF()

    # Two targets cannot depend on the same generated files for
    # parallel builds. Thus, if a module has IDL files, serialize
    # the build
    IF (NOT RTI_NO_SHARED_LIB)
        LIST(LENGTH ${RTI_MODULE_NAME}_SRC_IDL x)
        IF (${x} GREATER 0)
            ADD_DEPENDENCIES(${RTI_MODULE_LIBNAME_SHARED}
                             ${RTI_MODULE_LIBNAME_STATIC})
        ENDIF()
    ENDIF()

    foreach(md ${RTI_MODULE_IMPORTS} ${RTI_LIBRARY_DEPENDS})
        add_exported_header_dependency(DEPEDENCY
                                            ${md}
                                       DEPENDENTS
                                            ${RTI_MODULE_LIBNAME_STATIC}
        )
        if (NOT RTI_NO_SHARED_LIB)
            add_exported_header_dependency(DEPEDENCY
                                                ${md}
                                           DEPENDENTS
                                                ${RTI_MODULE_LIBNAME_SHARED}
            )
        endif()
    endforeach()

    IF (NOT RTI_NO_SHARED_LIB)
        LIST(LENGTH RTI_MODULE_DEPEND_SHARED_LIBS x)
        IF (${x} GREATER 0)
            ADD_DEPENDENCIES(${RTI_MODULE_LIBNAME_SHARED}
                             ${RTI_MODULE_DEPEND_SHARED_LIBS})
        ENDIF()
    ENDIF()

    IF (NOT RTI_NO_SHARED_LIB)
        # Do not import object dependencies
        FOREACH(mdin ${RTI_MODULE_IMPORTS} ${RTI_MODULE_DEPENDS}
        ${RTI_LIBRARY_DEPENDS} ${RTI_DLL_LIBRARY})
            STRING(REGEX REPLACE "lib(.*)" "\\1" md ${mdin})
            IF (${md} MATCHES "(.*)::(.*)")
                SET(md "${CMAKE_MATCH_1}")
                SET_PROPERTY(TARGET ${RTI_MODULE_LIBNAME_SHARED} APPEND
                PROPERTY COMPILE_DEFINITIONS RTI_${CMAKE_MATCH_2}_DLL_VARIABLE)
            ENDIF()
            IF (RTIME_PIL AND ("${md}" STREQUAL "osapi"))
                SET_PROPERTY(TARGET ${RTI_MODULE_LIBNAME_SHARED} APPEND
                PROPERTY COMPILE_DEFINITIONS RTI_ospsl_DLL_VARIABLE)
            ENDIF()
            SET_PROPERTY(TARGET ${RTI_MODULE_LIBNAME_SHARED} APPEND
            PROPERTY COMPILE_DEFINITIONS RTI_${md}_DLL_VARIABLE)
            IF (DEFINED ${mdin}_OBJECT_DEPENDS)
                FOREACH(mdin2 ${${mdin}_OBJECT_DEPENDS})
                    IF (${mdin2} MATCHES "(.*)::(.*)")
                        SET(mdin2 "${CMAKE_MATCH_1}")
                        SET_PROPERTY(TARGET ${RTI_MODULE_LIBNAME_SHARED} APPEND
                        PROPERTY COMPILE_DEFINITIONS RTI_${CMAKE_MATCH_2}_DLL_VARIABLE)
                    ENDIF()
                    IF (RTIME_PIL AND ("${mdin2}" STREQUAL "osapi"))
                        SET_PROPERTY(TARGET ${RTI_MODULE_LIBNAME_SHARED} APPEND
                        PROPERTY COMPILE_DEFINITIONS RTI_ospsl_DLL_VARIABLE)
                    ENDIF()
                    SET_PROPERTY(TARGET ${RTI_MODULE_LIBNAME_SHARED} APPEND
                        PROPERTY COMPILE_DEFINITIONS RTI_${mdin2}_DLL_VARIABLE)
                ENDFOREACH()
            ENDIF()
        ENDFOREACH()

        FOREACH(mdin ${RTI_MODULE_EXPORTS})
            STRING(REGEX REPLACE "lib(.*)" "\\1" md ${mdin})
            IF (${md} MATCHES "(.*)::(.*)")
                SET(md "${CMAKE_MATCH_1}")
            ENDIF()
            SET_PROPERTY(TARGET
                            ${RTI_MODULE_LIBNAME_SHARED}
                        APPEND PROPERTY COMPILE_DEFINITIONS
                            RTI_${md}_DLL_EXPORT
                            RTI_${md}_DLL_VARIABLE
            )
        ENDFOREACH()
    ENDIF()

    # For shared libraries the module must export its public functions.
    IF (NOT RTI_NO_SHARED_LIB)
        SET_PROPERTY(TARGET ${RTI_MODULE_LIBNAME_SHARED}  APPEND
                     PROPERTY
                     COMPILE_DEFINITIONS RTI_${RTI_MODULE_NAME}_DLL_EXPORT
                                         RTI_${RTI_MODULE_NAME}_DLL_VARIABLE)

        FOREACH (m ${RTI_OBJECT_DEPENDS})
            IF (${m} MATCHES "(.*)::(.*)")
                IF (RTI_${CMAKE_MATCH_2}_EXPORTS)
                    FOREACH (e ${RTI_${CMAKE_MATCH_2}_EXPORTS})
                        SET_PROPERTY(TARGET ${RTI_MODULE_LIBNAME_SHARED}  APPEND
                                     PROPERTY
                                     COMPILE_DEFINITIONS
                                        RTI_${e}_DLL_EXPORT
                                        RTI_${e}_DLL_VARIABLE
                        )
                    ENDFOREACH()
                ENDIF()
            ELSE()
                SET_PROPERTY(TARGET ${RTI_MODULE_LIBNAME_SHARED}  APPEND
                             PROPERTY
                             COMPILE_DEFINITIONS RTI_${m}_DLL_EXPORT
                                                 RTI_${m}_DLL_VARIABLE
                )
            ENDIF()
        ENDFOREACH()

        SET_PROPERTY(TARGET ${RTI_MODULE_LIBNAME_SHARED} APPEND
                            PROPERTY MACOSX_RPATH TRUE)
    ENDIF()

    TARGET_LINK_LIBRARIES(${RTI_MODULE_LIBNAME_STATIC})

    # Add dependencies for the shared libraries, including DLL variables
    IF (RTI_NO_SHARED_LIB)
        #MESSAGE("not linking ${RTI_MODULE_LIBNAME_SHARED}")
    ELSE()
        TARGET_LINK_LIBRARIES(${RTI_MODULE_LIBNAME_SHARED}
                              ${RTI_MODULE_DEPEND_SHARED_LIBS} ${PLATFORM_LIBS})

        SET_PROPERTY(TARGET ${RTI_MODULE_LIBNAME_SHARED}  APPEND
                     PROPERTY COMPILE_DEFINITIONS
                     RTIME_TARGET_NAME=\"${RTIME_TARGET_NAME}\"
        )
    ENDIF()

    SET_PROPERTY(TARGET ${RTI_MODULE_LIBNAME_STATIC}  APPEND
                 PROPERTY COMPILE_DEFINITIONS
                 RTIME_TARGET_NAME=\"${RTIME_TARGET_NAME}\"
    )

    IF (NOT ${RTIME_OUTPUT_LIBDIR_PREFIX} MATCHES ".*/$")
        SET(RTIME_OUTPUT_LIBDIR_PREFIX "${RTIME_OUTPUT_LIBDIR_PREFIX}/")
    ENDIF()
    # Export libraries

    IF (DEFINED RTI_MODULE_EXPORT OR RTI_MODULE_EXPORT_HEADER)
        IF (NOT RTI_NO_SHARED_LIB AND RTI_MODULE_EXPORT)
            # For windows we need to copy .exp,*.lib., *.pdb and *.dll
            IF (${CMAKE_SHARED_LIBRARY_SUFFIX} STREQUAL ".dll")
                SET(libsuffix "dll" "lib" "exp")
                IF (${RTI_DEBUG_BUILD})
                    LIST(APPEND libsuffix "pdb")
                ENDIF()
                FOREACH(f ${libsuffix})
                   ADD_CUSTOM_COMMAND(TARGET ${RTI_MODULE_LIBNAME_SHARED} POST_BUILD
                               COMMAND ${CMAKE_COMMAND} -E copy
                               $<TARGET_LINKER_FILE_DIR:${RTI_MODULE_LIBNAME_SHARED}>/${CMAKE_SHARED_LIBRARY_PREFIX}${RTI_MODULE_LIBNAME_SHARED}.${f}
                               ${RTI_EXPORT_MODULE_OUTPUT_LIB}/${RTIME_OUTPUT_LIBDIR_PREFIX}${CMAKE_SHARED_LIBRARY_PREFIX}${RTI_MODULE_LIBNAME_SHARED}.${f})
                ENDFOREACH()
            ELSE()
                ADD_CUSTOM_COMMAND(TARGET ${RTI_MODULE_LIBNAME_SHARED} POST_BUILD
                           COMMAND ${CMAKE_COMMAND} -E copy
                           $<TARGET_LINKER_FILE:${RTI_MODULE_LIBNAME_SHARED}>
                           ${RTI_EXPORT_MODULE_OUTPUT_LIB}/${RTIME_OUTPUT_LIBDIR_PREFIX}${CMAKE_SHARED_LIBRARY_PREFIX}${RTI_MODULE_LIBNAME_SHARED}${CMAKE_SHARED_LIBRARY_SUFFIX})
            ENDIF()
            IF (RTIME_HAS_DLL_MANIFEST_INPUT)
                SET(RTIME_HAS_DLL_MANIFEST_DEPEND_LIST "")
                STRING(REPLACE "PRODUCT" ${RTIME_PRODUCT_NAME} config_output_file ${RTIME_HAS_DLL_MANIFEST_OUTPUT})
                STRING(REPLACE "LIBRARY" ${RTI_MODULE_LIBNAME_SHARED} config_output_file  ${config_output_file})
                FOREACH(d ${RTI_MODULE_DEPEND_SHARED_LIBS})
                    STRING(REPLACE "PRODUCT" ${RTIME_PRODUCT_NAME} depend_line ${RTIME_HAS_DLL_MANIFEST_DEPEND_PATTERN})
                    STRING(REPLACE "LIBRARY" ${d} depend_line ${depend_line})
                    SET(RTIME_HAS_DLL_MANIFEST_DEPEND_LIST "${RTIME_HAS_DLL_MANIFEST_DEPEND_LIST}\n${depend_line}")
                ENDFOREACH()
                configure_file(${RTIME_HAS_DLL_MANIFEST_INPUT} ${RTI_EXPORT_MODULE_OUTPUT_LIB}/${config_output_file})
                UNSET(RTIME_HAS_DLL_MANIFEST_DEPEND_LIST)
            ENDIF()
        ENDIF()

        IF (NOT RTIME_EXCLUDE_STATIC_LIB_COPY AND RTI_MODULE_EXPORT)
            IF ((${RTI_MODULE_LIBNAME_STATIC} MATCHES "test") AND (NOT ${RTIME_DEVTREE}))
                ADD_CUSTOM_COMMAND(TARGET ${RTI_MODULE_LIBNAME_STATIC} POST_BUILD
                            COMMAND ${CMAKE_COMMAND} -E make_directory
                                    ${RTIME_MODULE_TEST_EXPORT}/bin/${RTIME_TARGET_NAME}
                            COMMAND ${CMAKE_COMMAND} -E copy
                                    $<TARGET_LINKER_FILE:${RTI_MODULE_LIBNAME_STATIC}>
                                    ${RTIME_MODULE_TEST_EXPORT}/bin/${RTIME_TARGET_NAME}/${RTIME_OUTPUT_LIBDIR_PREFIX}${CMAKE_STATIC_LIBRARY_PREFIX}${RTI_MODULE_LIBNAME_STATIC}${CMAKE_STATIC_LIBRARY_SUFFIX})
            ELSE()
                # This is just to test if this is a windows build, not if
                # this is a shared library.
                IF (${CMAKE_SHARED_LIBRARY_SUFFIX} STREQUAL ".dll")
                    SET(libsuffix "lib")
                    IF (${RTI_DEBUG_BUILD})
                        LIST(APPEND libsuffix "pdb")
                    ENDIF()
                    FOREACH(f ${libsuffix})
                       ADD_CUSTOM_COMMAND(TARGET ${RTI_MODULE_LIBNAME_STATIC} POST_BUILD
                                   COMMAND ${CMAKE_COMMAND} -E copy
                                   $<TARGET_LINKER_FILE_DIR:${RTI_MODULE_LIBNAME_STATIC}>/${CMAKE_STATIC_LIBRARY_PREFIX}${RTI_MODULE_LIBNAME_STATIC}.${f}
                                   ${RTI_EXPORT_MODULE_OUTPUT_LIB}/${RTIME_OUTPUT_LIBDIR_PREFIX}${CMAKE_STATIC_LIBRARY_PREFIX}${RTI_MODULE_LIBNAME_STATIC}.${f})
                    ENDFOREACH()
                ELSE()
                    ADD_CUSTOM_COMMAND(TARGET ${RTI_MODULE_LIBNAME_STATIC} POST_BUILD
                               COMMAND ${CMAKE_COMMAND} -E copy
                               $<TARGET_LINKER_FILE:${RTI_MODULE_LIBNAME_STATIC}>
                               ${RTI_EXPORT_MODULE_OUTPUT_LIB}/${RTIME_OUTPUT_LIBDIR_PREFIX}${CMAKE_STATIC_LIBRARY_PREFIX}${RTI_MODULE_LIBNAME_STATIC}${CMAKE_STATIC_LIBRARY_SUFFIX})
                ENDIF()
            ENDIF()
        ENDIF()

        SET(${RTI_MODULE_NAME}_EXPORT_GENERATED)
        IF (${RTIME_DEVTREE})
            # Export public header-files to the product directory
            FOREACH(file_name ${RTI_MODULE_PUBLIC_HDR} ${RTI_MODULE_SHARED_HDR_PATCHED})
                # include files prefixed with rti_me goes into the root include directory

                    SET(output_dir ${RTIME_MODULE_EXPORT_INCLUDE_ROOT})
                else()
                    set(output_dir ${RTIME_MODULE_EXPORT_INCLUDE_ROOT}/${RTI_MODULE_NAME})
                endif()
                add_header_export(
                    MODULE_NAME ${RTI_MODULE_NAME}
                    TARGET_LIB ${RTI_MODULE_LIBNAME_STATIC}
                    INPUT_FILE ${RTI_MODULE_SRCROOT}/include/${RTI_MODULE_NAME}/${file_name}
                    OUTPUT_FILE ${output_dir}/${file_name}

                list(APPEND ${RTI_MODULE_NAME}_EXPORT_GENERATED ${output_dir}/${file_name})
            ENDFOREACH()

                    SET(RTI_MODULE_SHARED_HDR_DIR include/share)
                    SET(RTI_SHARED_MODULE_EXPORT_DIR)

                    IF (EXISTS ${RTI_MODULE_SRCROOT}/include/${sd}/CMakeLists.txt)
                        INCLUDE(${RTI_MODULE_SRCROOT}/include/${sd}/CMakeLists.txt)
                    ENDIF()

                    IF (DEFINED RTI_MODULE_SHARED_HDR)
                        FOREACH(f ${RTI_MODULE_SHARED_HDR} ${RTI_MODULE_SHARED_HDR_PATCHED})
                            IF (DEFINED RTI_SHARED_MODULE_EXPORT_DIR)
                                set(output_dir ${RTIME_MODULE_EXPORT_INCLUDE_ROOT}/${RTI_SHARED_MODULE_EXPORT_DIR}${sd})

                            endif()
                            add_header_export(

                                    TARGET_LIB ${RTI_MODULE_LIBNAME_STATIC}
                                    INPUT_FILE ${RTI_MODULE_SRCROOT}/include/${sd}/${f}

                    ENDIF()

            FOREACH(idl ${${RTI_MODULE_NAME}_IDL_EXPORT})
                    GET_FILENAME_COMPONENT(idl_file ${idl} NAME)

                        MODULE_NAME ${RTI_MODULE_NAME}
                        TARGET_LIB ${RTI_MODULE_LIBNAME_STATIC}

                                  DEPENDS
                                    ${${RTI_MODULE_NAME}_EXPORT_GENERATED})
            ENDIF()
        ENDIF()
    ENDIF()
ELSEIF ((DEFINED RTI_BUILDID_MASTER) AND (NOT RTIME_DISABLE_BUILDID))

        #LIST(APPEND RTI_SHARED_OBJLIB ${BUILD_ID_FILE})

        # When there are multiple output files, CMake generated a dependency
        # between them based on the order in which they appear. So the below
        # creates a rule where BUILD_ID_FILE depends on PHONY_BUILDID.
        # However, PHONY_BUILDID generates BUILD_ID_FILE so it is always
        # out of date.
        ADD_CUSTOM_COMMAND(OUTPUT  ${RTI_MODULE_OUTPUT_INCLUDE}
                           COMMAND ${CMAKE_COMMAND} -E make_directory ${RTI_MODULE_OUTPUT_INCLUDE}
                           COMMENT "Generate ${RTI_MODULE_OUTPUT_INCLUDE}"
        )

        IF (RTI_MANUAL_BUILDID)
            ADD_CUSTOM_COMMAND(OUTPUT  ${BUILD_ID_FILE} ${RTI_MODULE_OUTPUT_INCLUDE}
                               COMMAND ${CMAKE_COMMAND} -DBUILD_ID_FILE=${BUILD_ID_FILE} -DHAS_STRING_TIMESTAMP=${HAS_STRING_TIMESTAMP} -P ${RTIME_RESOURCE_ROOT}/cmake/cmake_buildid.txt
                               COMMENT "Generate ${BUILD_ID_FILE}"
            )
            ADD_CUSTOM_TARGET(BUILDID
                              DEPENDS ${BUILD_ID_FILE}
            )
        ELSE()
            ADD_CUSTOM_COMMAND(OUTPUT  PHONY_BUILDID ${BUILD_ID_FILE} ${RTI_MODULE_OUTPUT_INCLUDE}
                               COMMAND ${CMAKE_COMMAND} -DBUILD_ID_FILE=${BUILD_ID_FILE} -DHAS_STRING_TIMESTAMP=${HAS_STRING_TIMESTAMP} -P ${RTIME_RESOURCE_ROOT}/cmake/cmake_buildid.txt
                               COMMENT "Generate ${BUILD_ID_FILE}"
            )
            ADD_CUSTOM_TARGET(BUILDID
                              DEPENDS PHONY_BUILDID
            )
        ENDIF()

        SET_SOURCE_FILES_PROPERTIES( PHONY_BUILDID PROPERTIES SYMBOLIC 1)
ELSEIF ((DEFINED RTI_BUILDID_MASTER) AND (DEFINED RTIME_DISABLE_BUILDID))
        ADD_CUSTOM_TARGET(BUILDID  )
