IF (RTIME_PRINT_CONFIG)
    RETURN()
ENDIF()

SET(RTIME_CC_AUTO    "<auto detect>")
SET(RTIME_CC_DEFAULT ${RTIME_CC_AUTO})

# If the compiler file is already specified, just add
IF (DEFINED RTIME_OSAPI_COMPILER AND NOT ${RTIME_OSAPI_COMPILER} MATCHES ${RTIME_CC_AUTO})
    IF (NOT OSAPI_CC_DEF_H_SET)
        ADD_DEFINITIONS(-DOSAPI_CC_DEF_H=osapi/${RTIME_OSAPI_COMPILER})
    ENDIF()
    RETURN()
ENDIF()

INCLUDE(TestBigEndian)

#== Look for available platforms
SET(RTIME_OSAPI_MODULE "osapi.2.0")

GET_MODULE_DIR(${RTIME_OSAPI_MODULE} osdir)

IF (DEFINED _RTIME_CC)
    SET(RTIME_CC_DEFAULT "osapi_cc_${_RTIME_CC}.h")
ENDIF()

SET(RTIME_OSAPI_COMPILER_DOC 
"Select the C/C++ Compiler definitions. The stdc supports GCC and MSVC")

SET(RTIME_OSAPI_COMPILER ${RTIME_CC_DEFAULT} CACHE STRING ${RTIME_OSAPI_COMPILER_DOC})

LIST(APPEND RTIME_OSAPI_COMPILER_LIST ${RTIME_CC_AUTO})

IF (${RTIME_DEVTREE})
    FILE(GLOB compilers LIST_DIRECTORIES FALSE 
        ${RTIME_SOURCE_ROOT}/${osdir}/include/osapi/*osapi_cc_*.h)
ELSE()
    FILE(GLOB compilers LIST_DIRECTORIES FALSE 
         ${RTIME_SOURCE_ROOT}/include/osapi/*osapi_cc_*.h)
ENDIF()

IF (${RTIME_OSAPI_COMPILER} MATCHES ${RTIME_CC_AUTO})
    SET(stdc_FILE osapi_cc_stdc.h)
    SET(RTIME_OSAPI_COMPILER_SELECT stdc CACHE
        STRING "Default compiler selection")

    FOREACH(x ${compilers})
            GET_FILENAME_COMPONENT(ccname ${x} NAME)
            STRING(REGEX REPLACE ".*osapi_cc_(.*).h" "\\1" cc ${ccname})
            LIST(APPEND RTIME_OSAPI_COMPILER_LIST ${cc})
            SET(${cc}_FILE ${ccname})
    ENDFOREACH()

    SET_PROPERTY(CACHE RTIME_OSAPI_COMPILER_SELECT PROPERTY
                 STRINGS ${RTIME_OSAPI_COMPILER_LIST})

    SET(RTIME_OSAPI_COMPILER ${${RTIME_OSAPI_COMPILER_SELECT}_FILE})
    IF (NOT OSAPI_CC_DEF_H_SET)
        ADD_DEFINITIONS(
            -DOSAPI_CC_DEF_H=osapi/${${RTIME_OSAPI_COMPILER_SELECT}_FILE}
        )
    ENDIF()

ELSE()
    IF (NOT OSAPI_CC_DEF_H_SET)
        ADD_DEFINITIONS(-DOSAPI_CC_DEF_H=osapi/${RTIME_OSAPI_COMPILER})
    ENDIF()
ENDIF()

# This sets the compiler byte order. It might be needed for
# old gcc versions which do not define __BYTE_ORDER__
IF (NOT DEFINED RTI_ENDIAN_BIG AND NOT DEFINED RTI_ENDIAN_LITTLE)
    TEST_BIG_ENDIAN(RTI_CMAKE_TEST_ENDIAN)
    IF (DEFINED RTI_CMAKE_TEST_ENDIAN)
        IF (${RTI_CMAKE_TEST_ENDIAN})
            ADD_DEFINITIONS(-DRTI_ENDIAN_BIG=1)
        ELSE()
            ADD_DEFINITIONS(-DRTI_ENDIAN_LITTLE=1)
        ENDIF()
    ENDIF()
ELSEIF(DEFINED RTI_ENDIAN_BIG)
    ADD_DEFINITIONS(-DRTI_ENDIAN_BIG=1)
ELSEIF(DEFINED RTI_ENDIAN_LITTLE)
    ADD_DEFINITIONS(-DRTI_ENDIAN_LITTLE=1)
ENDIF()
