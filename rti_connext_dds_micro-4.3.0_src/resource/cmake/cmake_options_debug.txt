IF (RTIME_PRINT_CONFIG)
    RTI_MESSAGE_STATUS("RTIME_OSAPI_ENABLE_LOG .............: ${RTIME_OSAPI_ENABLE_LOG}")
    RTI_MESSAGE_STATUS("RTIME_OSAPI_ENABLE_PRECONDITION ....: ${RTIME_OSAPI_ENABLE_PRECONDITION}")
    RETURN()
ENDIF()

#== Debug information
IF (NOT RTI_DEBUG_BUILD AND RTIME_CERT)
    SET(RTIME_INCLUDE_LOG FALSE)
ELSE()
    SET(RTIME_INCLUDE_LOG TRUE)
ENDIF()

LIST(APPEND DEBUG_OPTIONS "Include" "Exclude" "Default")

SET(RTIME_OSAPI_ENABLE_LOG "Default" CACHE STRING "Include/Exclude Logging")
SET_PROPERTY(CACHE RTIME_OSAPI_ENABLE_LOG PROPERTY STRINGS ${DEBUG_OPTIONS})

SET(RTIME_OSAPI_ENABLE_TRACE "Default" CACHE STRING "Include/Exclude Trace information")
SET_PROPERTY(CACHE RTIME_OSAPI_ENABLE_TRACE PROPERTY STRINGS ${DEBUG_OPTIONS})

SET(RTIME_OSAPI_ENABLE_PRECONDITION "Default" CACHE STRING "Include/Exclude Precondition checks")
SET_PROPERTY(CACHE RTIME_OSAPI_ENABLE_PRECONDITION PROPERTY STRINGS ${DEBUG_OPTIONS})

IF (${RTIME_OSAPI_ENABLE_TRACE} STREQUAL "Include")
    MESSAGE("Update RTIME_OSAPI_ENABLE_LOG") 
    SET_PROPERTY(CACHE RTIME_OSAPI_ENABLE_LOG PROPERTY VALUE "Include")
    MESSAGE("Update RTIME_OSAPI_ENABLE_LOG=${RTIME_OSAPI_ENABLE_LOG}") 
ENDIF()

IF (${RTIME_OSAPI_ENABLE_LOG} STREQUAL "Include")
    SET(RTIME_INCLUDE_LOG TRUE)
ELSEIF(${RTIME_OSAPI_ENABLE_LOG} STREQUAL "Exclude")
    SET(RTIME_INCLUDE_LOG FALSE)
ENDIF()

IF (${RTIME_OSAPI_ENABLE_TRACE} STREQUAL "Include")
    ADD_DEFINITIONS(-DOSAPI_ENABLE_TRACE=1)
ELSEIF(${RTIME_OSAPI_ENABLE_LOG} STREQUAL "Exclude")
    ADD_DEFINITIONS(-DOSAPI_ENABLE_TRACE=0)
ENDIF()

IF (${RTIME_OSAPI_ENABLE_PRECONDITION} STREQUAL "Include")
    ADD_DEFINITIONS(-DOSAPI_ENABLE_PRECONDITION=1)
ELSEIF(${RTIME_OSAPI_ENABLE_LOG} STREQUAL "Exclude")
    ADD_DEFINITIONS(-DOSAPI_ENABLE_PRECONDITION=0)
ENDIF()

IF (RTIME_INCLUDE_LOG)
    ADD_DEFINITIONS(-DOSAPI_ENABLE_LOG=1)
ELSE()
    ADD_DEFINITIONS(-DOSAPI_ENABLE_LOG=0)
ENDIF()
