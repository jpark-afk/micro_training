IF (RTIME_PRINT_CONFIG)
    RTI_MESSAGE_STATUS("RTIME_UDP_EXCLUDE_BUILTIN ..........: ${RTIME_UDP_EXCLUDE_BUILTIN}")
    IF (RTIME_UDP_EXCLUDE_BUILTIN)
        RETURN()
    ENDIF()
    RTI_MESSAGE_STATUS("RTIME_UDP_ENABLE_IPALIASES .........: ${RTIME_UDP_ENABLE_IPALIASES}")
    RTI_MESSAGE_STATUS("RTIME_UDP_ENABLE_TRANSFORMS ........: ${RTIME_UDP_ENABLE_TRANSFORMS}")
    RTI_MESSAGE_STATUS("RTIME_UDP_EXCLUDE_BUILTIN ..........: ${RTIME_UDP_EXCLUDE_BUILTIN}")
    RTI_MESSAGE_STATUS("RTIME_UDP_ENABLE_MULTICAST .........: ${RTIME_UDP_ENABLE_MULTICAST}")
    RTI_MESSAGE_STATUS("RTIME_UDP_ENFORCE_TRANSPORT_PRIORITY .........: ${RTIME_UDP_ENFORCE_TRANSPORT_PRIORITY}")
    RETURN()
ENDIF()

IF (RTIME_UDP_EXCLUDE_BUILTIN)
    ADD_DEFINITIONS(-DUDP_EXCLUDE_BUILTIN=1)
    RETURN()
ENDIF()

# This option is used to select whether or not IP aliases are filtered out.
# Checking this option disables filtering out IP aliases. Note that this only
# works on platforms where each IP alias has its own interface name, such
# as eth0:1, eth0:1 etc.
SET(RTIME_UDP_ENABLE_IPALIASES_DOC
"Checking this option disables filtering out IP aliases. Note that this only
works on platforms where each IP alias has its own interface name, such
as eth0:1, eth0:2 etc")

SET(RTIME_UDP_ENABLE_IPALIASES FALSE CACHE BOOL ${RTIME_UDP_ENABLE_IPALIASES_DOC})

IF (${RTIME_UDP_ENABLE_IPALIASES})
    ADD_DEFINITIONS(-DUDP_ENABLE_IPALIASES=1)
ENDIF()

# This option enables the UDP transformation path in the UDP transport.
SET(RTIME_UDP_ENABLE_TRANSFORMS_DOC
"Checking this option enables UDP Transformations in the UDP transport")

SET(RTIME_UDP_ENABLE_TRANSFORMS FALSE CACHE BOOL ${RTIME_UDP_ENABLE_TRANSFORMS_DOC})

IF (${RTIME_UDP_ENABLE_TRANSFORMS})
    ADD_DEFINITIONS(-DUDP_ENABLE_TRANSFORMS=1)
ENDIF()

# This option enables UDP interface bind for UDP sockets.
SET(RTIME_UDP_ENABLE_INTERFACE_BIND_DOC
"Checking this option enables binding a UDP port to a specific interface")

IF (RTIME_CERT)
    SET(RTIME_UDP_ENABLE_INTERFACE_BIND FALSE CACHE BOOL ${RTIME_UDP_ENABLE_INTERFACE_BIND_DOC})
ELSE()
    SET(RTIME_UDP_ENABLE_INTERFACE_BIND TRUE CACHE BOOL ${RTIME_UDP_ENABLE_INTERFACE_BIND_DOC})
ENDIF()

IF (${RTIME_UDP_ENABLE_INTERFACE_BIND})
    ADD_DEFINITIONS(-DUDP_ENABLE_INTERFACE_BIND=1)
ENDIF()

# This option removes the UDP transport from being built. This can be useful if
# communication is done only using shared memory, INTRA or a custom UDP transport.
SET(RTIME_UDP_EXCLUDE_BUILTIN_DOC
"Checking this option removes UDP transport from being built")

IF (NOT DEFINED RTIME_UDP_EXCLUDE_BUILTIN)
    SET(RTIME_UDP_EXCLUDE_BUILTIN FALSE CACHE BOOL ${RTIME_UDP_EXCLUDE_BUILTIN_DOC})
ELSE()
    SET(RTIME_UDP_EXCLUDE_BUILTIN ${RTIME_UDP_EXCLUDE_BUILTIN} CACHE BOOL ${RTIME_UDP_EXCLUDE_BUILTIN_DOC})
ENDIF()

IF (${RTIME_UDP_EXCLUDE_BUILTIN})
    ADD_DEFINITIONS(-DUDP_EXCLUDE_BUILTIN=1)
ENDIF()

# This option enables/disables multicast
SET(RTIME_UDP_ENABLE_MULTICAST_DOC
"Checking this option enables multicast")

IF (NOT DEFINED RTIME_UDP_ENABLE_MULTICAST)
    SET(RTIME_UDP_ENABLE_MULTICAST TRUE CACHE BOOL ${RTIME_UDP_ENABLE_MULTICAST_DOC})
ELSE()
    SET(RTIME_UDP_ENABLE_MULTICAST ${RTIME_UDP_ENABLE_MULTICAST} CACHE BOOL ${RTIME_UDP_ENABLE_MULTICAST_DOC})
ENDIF()

IF (${RTIME_UDP_ENABLE_MULTICAST})
    ADD_DEFINITIONS(-DNETIO_CONFIG_ENABLE_MULTICAST=1)
ELSE()
    ADD_DEFINITIONS(-DNETIO_CONFIG_ENABLE_MULTICAST=0)
ENDIF()

# This option enables/disables transport priority enforcement
SET(RTIME_UDP_ENFORCE_TRANSPORT_PRIORITY_DOC
"Checking this option enables transport priority enforcement")

IF (NOT DEFINED RTIME_UDP_ENFORCE_TRANSPORT_PRIORITY)
    SET(RTIME_UDP_ENFORCE_TRANSPORT_PRIORITY TRUE CACHE BOOL ${RTIME_UDP_ENFORCE_TRANSPORT_PRIORITY_DOC})
ELSE()
    SET(RTIME_UDP_ENFORCE_TRANSPORT_PRIORITY ${RTIME_UDP_ENFORCE_TRANSPORT_PRIORITY} CACHE BOOL ${RTIME_UDP_ENFORCE_TRANSPORT_PRIORITY_DOC})
ENDIF()

IF (${RTIME_UDP_ENFORCE_TRANSPORT_PRIORITY})
    ADD_DEFINITIONS(-DUDP_ENFORCE_TRANSPORT_PRIORITY=1)
ELSE()
    ADD_DEFINITIONS(-DUDP_ENFORCE_TRANSPORT_PRIORITY=0)
ENDIF()
